```go
// _scaffold routes (D-46). Phase 1's hand-written (w,r) handlers are
// migrated to typed strict-server handlers; Phase 3 deletes both the
// package and the corresponding openapi.yaml paths when real catalog
// entities replace _scaffold.
//
// Invariants carried forward from Phase 1 (MUST NOT regress — FOUND-08
// integration suite proves them):
//   - Authoritative org_id from ctx (FOUND-05, never URL/body)
//   - UUIDv7 minted server-side at write boundary (D-19)
//   - pgx.ErrNoRows → 404 with "not_found" code (FOUND-08 leakage guard)
package scaffold

import (
    "context"
    "errors"

// Handler implements the Scaffold portion of api.StrictServerInterface
// (D-46). Other Plan-04 work composes this with a "not implemented" handler
// that fills Agents/Skills/Queues/Channels/Adapters/BreakReasons/AgentStates/
// Imports — see server.go.
//
// NOTE: handlers do NOT populate the ErrorResponse.RequestId field — that
// concern is centralized in server.requestIDInjectionMiddleware (B-1, D-35),
// a StrictMiddlewareFunc that wraps every operation and injects request_id
// from ctx into any returned ErrorResponse-shaped body. Handlers focus on
// business logic only.
type Handler struct {
    orgDB *db.OrgDB
}

func NewHandler(orgDB *db.OrgDB) *Handler {
    return &Handler{orgDB: orgDB}
}

// CreateScaffold implements api.StrictServerInterface.CreateScaffold.
// Spec contract: POST /v1/orgs/{org_id}/_scaffold with {external_id, name}
// returns 201 Scaffold (CAT-like — but _scaffold is throwaway).
//
// codegen has already JSON-decoded the body into req.Body and rejected
// missing required fields per openapi.yaml. The handler focuses on
// org_id resolution + UUIDv7 mint + DB insert.
func (h *Handler) CreateScaffold(ctx context.Context, req api.CreateScaffoldRequestObject) (api.CreateScaffoldResponseObject, error) {
    orgID, ok := orgkey.OrgIDFromContext(ctx)
    if !ok {
        // Defensive guard: OrgContext middleware guarantees presence inside
        // /v1; this branch only fires on a wiring regression we want to
        // surface loudly. RequestId field stays empty here; server-side
        // requestIDInjectionMiddleware fills it on the way out.
        return api.CreateScaffold500JSONResponse(api.ErrorResponse{
            Error:  api.Internal,
            Reason: "missing_org_id_in_context",
        }), nil
    }

    q := generated.New(h.orgDB)
    id := uuid.Must(uuid.NewV7()) // D-19
    row, err := q.InsertScaffold(ctx, generated.InsertScaffoldParams{
        ID:         toPgUUID(id),
        OrgID:      toPgUUID(orgID),
        ExternalID: req.Body.ExternalId, // codegen field name from openapi `external_id`
        Name:       req.Body.Name,
    })
    if err != nil {
        return api.CreateScaffold500JSONResponse(api.ErrorResponse{
            Error:  api.Internal,
            Reason: "insert_failed",
        }), nil
    }
    return api.CreateScaffold201JSONResponse(toAPIScaffold(row)), nil
}

// ListScaffolds implements api.StrictServerInterface.ListScaffolds.
func (h *Handler) ListScaffolds(ctx context.Context, req api.ListScaffoldsRequestObject) (api.ListScaffoldsResponseObject, error) {
    orgID, ok := orgkey.OrgIDFromContext(ctx)
    if !ok {
        return api.ListScaffolds500JSONResponse(api.ErrorResponse{
            Error:  api.Internal,
            Reason: "missing_org_id_in_context",
        }), nil
    }
    q := generated.New(h.orgDB)
    rows, err := q.ListScaffolds(ctx, toPgUUID(orgID))
    if err != nil {
        return api.ListScaffolds500JSONResponse(api.ErrorResponse{
            Error:  api.Internal,
            Reason: "list_failed",
        }), nil
    }
    out := make([]api.Scaffold, 0, len(rows))
    for _, s := range rows {
        out = append(out, toAPIScaffold(s))
    }
    return api.ListScaffolds200JSONResponse(out), nil
}

// GetScaffoldById implements api.StrictServerInterface.GetScaffoldById.
// FOUND-08 cross-org probe: a row in another org returns pgx.ErrNoRows
// because the sqlc query body filters by (id, org_id); the handler maps
// that to 404 — identical to a genuinely-missing row.
func (h *Handler) GetScaffoldById(ctx context.Context, req api.GetScaffoldByIdRequestObject) (api.GetScaffoldByIdResponseObject, error) {
    orgID, ok := orgkey.OrgIDFromContext(ctx)
    if !ok {
        return api.GetScaffoldById500JSONResponse(api.ErrorResponse{
            Error:  api.Internal,
            Reason: "missing_org_id_in_context",
        }), nil
    }
    // codegen has already parsed the {id} path parameter (openapi format: uuid)
    // into req.Id (uuid.UUID); the handler does NOT call uuid.Parse manually.
    q := generated.New(h.orgDB)
    row, err := q.GetScaffoldByID(ctx, generated.GetScaffoldByIDParams{
        ID:    toPgUUID(req.Id),
        OrgID: toPgUUID(orgID),
    })
    if err != nil {
        if errors.Is(err, pgx.ErrNoRows) {
            return api.GetScaffoldById404JSONResponse(api.ErrorResponse{
                Error:  api.NotFound,
                Reason: "no_such_scaffold",
            }), nil
        }
        return api.GetScaffoldById500JSONResponse(api.ErrorResponse{
            Error:  api.Internal,
            Reason: "get_failed",
        }), nil
    }
    return api.GetScaffoldById200JSONResponse(toAPIScaffold(row)), nil
}

// toPgUUID + toAPIScaffold helpers. The Phase 1 toResponse(s) → scaffoldResponse
// helper is REMOVED — the strict-server returns api.Scaffold (codegen-defined)
// directly.

func toPgUUID(id uuid.UUID) pgtype.UUID {
    return pgtype.UUID{Bytes: id, Valid: true}
}

func toAPIScaffold(s generated.Scaffold) api.Scaffold {
    // pgtype.UUID -> uuid.UUID by reading Bytes directly (Valid is always true
    // for NOT NULL columns; sqlc emits pgtype to preserve nullable awareness).
    var createdAt time.Time
    if s.CreatedAt.Valid {
        createdAt = s.CreatedAt.Time
    }
    return api.Scaffold{
        Id:         uuid.UUID(s.ID.Bytes),
        OrgId:      uuid.UUID(s.OrgID.Bytes),
        ExternalId: s.ExternalID,
        Name:       s.Name,
        CreatedAt:  createdAt,
    }
}
```

```go
// RequestIDInjectionMiddleware is the StrictMiddlewareFunc that closes the
// D-35 gap for the strict-server pipeline (B-1). The strict pipeline
// bypasses middleware.WriteError, so without this wrapper any
// ErrorResponse the handler returns ships with an empty request_id — a
// regression from the Phase 1 envelope contract.
//
// The middleware runs AFTER the handler returns. It inspects the response
// object; if it embeds an api.ErrorResponse (or is convertible to one) and
// the RequestId field is empty, the middleware overrides RequestId with
// the value pulled from ctx (set by appmw.RequestID at the chi root).
// Successful responses (api.Scaffold, api.AgentList, etc.) pass through
// untouched.
//
// Exported so scaffold/handler_test.go can mount the same middleware in
// unit tests, proving the B-1 behavior at the package level (not only
// integration).
func RequestIDInjectionMiddleware() api.StrictMiddlewareFunc {
    return func(next api.StrictHandlerFunc, operationID string) api.StrictHandlerFunc {
        return func(ctx context.Context, w http.ResponseWriter, r *http.Request, request any) (any, error) {
            response, err := next(ctx, w, r, request)
            if err != nil || response == nil {
                return response, err
            }
            id, ok := appmw.RequestIDFromContext(ctx)
            if !ok {
                return response, nil
            }
            // Inject via reflection so we cover every per-operation
            // <Op><Status>JSONResponse type that wraps api.ErrorResponse
            // (the codegen names differ per operation, but all carry the
            // same {Error, Reason, RequestId} fields).
            injectRequestIDIntoErrorResponse(response, id.String())
            return response, nil
        }
    }
}

// injectRequestIDIntoErrorResponse looks for a `RequestId` string field on
// `response` (any *<Op><Status>JSONResponse the strict-server emits whose
// underlying type is api.ErrorResponse) and writes id when the field is
// empty. Uses reflect because oapi-codegen names the wrapper types per
// operation; there is no common interface for "this is an error envelope".
func injectRequestIDIntoErrorResponse(response any, id string) {
    v := reflect.ValueOf(response)
    // Strict-server response objects are typically named types whose
    // kind is Struct (e.g., type GetScaffoldById404JSONResponse ErrorResponse).
    // The wrapper is value-typed; dereference if pointer.
    for v.Kind() == reflect.Ptr {
        if v.IsNil() {
            return
        }
        v = v.Elem()
    }
    if v.Kind() != reflect.Struct {
        return
    }
    field := v.FieldByName("RequestId")
    if !field.IsValid() || !field.CanSet() || field.Kind() != reflect.String {
        return
    }
    if field.String() != "" {
        return  // handler already populated; respect that
    }
    field.SetString(id)
}
```

Imports added by this block: `"context"`, `"net/http"`, `"reflect"`, `appmw "github.com/luongdev/open-routing/services/api/internal/middleware"` (likely already present), `"github.com/luongdev/open-routing/services/api/internal/api"`.

NOTE on reflect cost: this runs once per request response; the reflect cost is ~50ns per call on modern hardware. Acceptable per ASVS V11.1.1 performance disposition (not a hot path).

NOTE on the strict-server response object shape: oapi-codegen v2 emits each `<Op><Status>JSONResponse` as a named type whose underlying type is the schema (`type GetScaffoldById404JSONResponse ErrorResponse`). The `reflect.ValueOf(...).Kind()` is `Struct`; calling `FieldByName("RequestId")` returns a settable field because the response value flows through the StrictMiddlewareFunc by interface — but the returned `response any` from the handler is a value (not a pointer) under the typical strict-server convention. If the installed oapi-codegen version returns pointer-typed responses, the `for v.Kind() == reflect.Ptr` loop above unwraps them. If `CanSet()` returns false because the value was passed by value (immutable), copy-modify-return: rebuild the response by constructing the same wrapper type with the new RequestId. The simpler `reflect.Value.Set` path covers value-typed strict-server output that oapi-codegen v2 emits today; if the executor finds the value is not settable, fall back to copy-modify-return per operation (the SUMMARY records the choice).

Alternative implementation if reflect proves brittle: emit a type-switch over EVERY `<Op><Status>JSONResponse` whose underlying type is `api.ErrorResponse`. This is mechanical but verbose (~37 cases for Plan 04's stub coverage). Prefer the reflect approach unless it produces incorrect behavior in unit tests; the SUMMARY records which path was used.
