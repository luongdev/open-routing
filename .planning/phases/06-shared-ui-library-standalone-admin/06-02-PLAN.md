    Create web/packages/ui/src/api/import.ts:

    This file is the SOLE gateway for POST /v1/orgs/{orgId}/catalog/import. It wraps raw fetch() internally and exposes a fully typed signature. No component or page file may call fetch() directly for this endpoint (ADMIN-04 enforcement — see acceptance_criteria grep gate).

    Imports:
    - import type { components } from './generated';
    - import { ApiError } from './errors';

    Type exports:
    - export type BulkImportResult = components['schemas']['BulkImportResult'];
    - export type CatalogEntity = 'agents' | 'skills' | 'queues' | 'channels' | 'adapters' | 'break_reasons';
    - export type ImportContentType = 'application/json' | 'text/csv';

    export interface ImportCatalogArgs {
      entity: CatalogEntity;
      body: string;            // raw JSON string or raw CSV text
      contentType: ImportContentType;
      schemaVersion?: 'v0.1';  // CSV only; ignored for JSON
      idempotencyKey?: string; // sets Idempotency-Key header per Phase 5 D5-13
    }

    export interface ImportCatalogDeps {
      baseURL: string;
      getOrgId: () => string;  // same source-of-truth as createApiClient (D-38)
      fetchImpl?: typeof fetch; // injectable for test; defaults to globalThis.fetch
    }

    export function createImporter(deps: ImportCatalogDeps) {
      const f = deps.fetchImpl ?? fetch;
      return async function importCatalog(args: ImportCatalogArgs): Promise<BulkImportResult> {
        const orgId = deps.getOrgId();
        const params = new URLSearchParams({ entity: args.entity });
        if (args.contentType === 'text/csv' && args.schemaVersion) {
          params.set('schema_version', args.schemaVersion);
        }
        const headers: Record<string, string> = {
          'Content-Type': args.contentType,
          'X-Org-Id': orgId,
          Accept: 'application/json',
        };
        if (args.idempotencyKey) headers['Idempotency-Key'] = args.idempotencyKey;

        const resp = await f(
          `${deps.baseURL}/v1/orgs/${orgId}/catalog/import?${params.toString()}`,
          { method: 'POST', headers, body: args.body }
        );
        const data = await resp.json().catch(() => ({}));
        if (resp.status >= 400 && resp.status !== 207) {
          throw new ApiError(resp.status, data);
        }
        return data as BulkImportResult; // 200 success OR 207 partial — caller renders both
      };
    }
