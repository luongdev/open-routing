<!-- Status API shape from Phase 4 -->
```typescript
type AgentStatus = 'Ready' | 'NotReady' | 'Break' | 'Engaged' | 'WrapUp' | 'Offline';

interface AgentStatusResponse {
  agent_id: string;
  status: AgentStatus;
  state_version: number;
  break_reason_id?: string | null;
  break_reason_name?: string | null;
  post_interaction_state?: 'ready' | 'not_ready' | null;
  wrapup_until?: string | null;  // ISO datetime string; null when not in WrapUp
  updated_at: string;
}

// PATCH /v1/orgs/{orgId}/agents/{id}/status body
interface PatchAgentStatusRequest {
  to: AgentStatus;
  break_reason_id?: string | null;   // required when to='Break'
  post_interaction_state?: 'ready' | 'not_ready';
  force?: boolean;  // admin override — logged at server (Phase 4 D-84)
}
```
