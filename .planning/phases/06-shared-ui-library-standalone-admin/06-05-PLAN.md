    @state() private _includeDisabled = false
    @state() private _limit = 25

    private _listTask = new Task(this, {
      task: async ([orgId, search, cursor, includeDisabled, limit]) => {
        const { data, error } = await this.client.GET('/v1/orgs/{org_id}/agents', {
          params: { path: { org_id: orgId }, query: { name: search || undefined, cursor: cursor ?? undefined, include_disabled: includeDisabled, limit } }
        });
        if (error) throw error;
        return data;
      },
      args: () => [this.orgId, this._search, this._cursor, this._includeDisabled, this._limit] as const,
    });
