```typescript
interface Queue {
  id: string;
  org_id: string;
  code: string;           // read-only post-create
  name: string;
  external_id?: string;
  channel_types: ('voice' | 'chat' | 'email')[];  // required, minItems=1
  priority: number;       // integer, required
  acw_sec: number;        // integer, required ("after-call work seconds")
  enabled: boolean;
  version: number;
  created_at: string;
  updated_at: string;
}
interface CreateQueueRequest {
  code: string; name: string; external_id?: string;
  channel_types: ('voice' | 'chat' | 'email')[];
  priority: number; acw_sec: number; enabled?: boolean;
}
interface UpdateQueueRequest {
  name?: string; external_id?: string;
  channel_types?: ('voice' | 'chat' | 'email')[];
  priority?: number; acw_sec?: number; enabled?: boolean;
  version: number;
}
```

| acw_sec integer → API | Non-negative integer; server rejects negative values with 422 |
