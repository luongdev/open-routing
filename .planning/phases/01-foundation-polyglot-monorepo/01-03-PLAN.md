    ```go
    package db

    import (
        "context"
        "fmt"
        "time"

        "github.com/jackc/pgx/v5/pgxpool"
    )

    // NewPool opens a pgxpool with retry-with-backoff against the dsn.
    // Retries up to 10 times with exponential backoff capped at 10s; total
    // worst-case wait ~50s, accommodating cold docker-compose startup (D-26).
    func NewPool(ctx context.Context, dsn string) (*pgxpool.Pool, error) {
        cfg, err := pgxpool.ParseConfig(dsn)
        if err != nil {
            return nil, fmt.Errorf("orgdb: parse dsn: %w", err)
        }
        var pool *pgxpool.Pool
        backoff := time.Second
        var lastErr error
        for attempt := 0; attempt < 10; attempt++ {
            pool, err = pgxpool.NewWithConfig(ctx, cfg)
            if err == nil {
                if perr := pool.Ping(ctx); perr == nil {
                    return pool, nil
                } else {
                    lastErr = perr
                    pool.Close()
                }
            } else {
                lastErr = err
            }
            select {
            case <-ctx.Done():
                return nil, ctx.Err()
            case <-time.After(backoff):
            }
            if backoff < 10*time.Second {
                backoff *= 2
                if backoff > 10*time.Second {
                    backoff = 10 * time.Second
                }
            }
        }
        return nil, fmt.Errorf("orgdb: pool unreachable after 10 attempts: %w", lastErr)
    }
    ```
    Notes:
    - Ctx cancellation is honored in the wait loop (`select` against `<-ctx.Done()`) so a shutdown signal short-circuits the retry.
    - The lastErr is preserved and wrapped in the final error so debugging reveals the actual Postgres failure (e.g., "connection refused" vs "FATAL: database does not exist").
    - This is the only file in the `internal/db` Go package created in this task — sqlcheck.go, bypass.go, orgdb.go land in Tasks 2 and 4 of this plan.

    File 2 — `services/api/internal/db/sqlcheck.go`. SQL inspector using pg_query_go AST walker per PATTERNS.md L5 lines 382-392 and Research §Pitfall 2 lines 1012-1020. Phase 1 walks the four DML statement types and accepts DDL (no org_id needed; bypass path handles migrations).
    ```go
    package db

    import (
        "crypto/sha256"
        "encoding/hex"
        "errors"
        "fmt"
        "strings"
        "sync"

        pg_query "github.com/pganalyze/pg_query_go/v6"
    )

    var ErrSQLMissingOrgFilter = errors.New("orgdb: SQL string missing org_id filter")

    // SQLChecker holds a cache of parsed SQL strings keyed by SHA-256 hash (D-02).
    type SQLChecker struct {
        mu    sync.RWMutex
        cache map[string]error // nil = accepted; ErrSQLMissingOrgFilter = rejected
    }

    func NewSQLChecker() *SQLChecker {
        return &SQLChecker{cache: make(map[string]error)}
    }

    // MustContainOrgFilter returns nil when the SQL string references org_id in an
    // appropriate position, or ErrSQLMissingOrgFilter when it does not.
    // Result is cached by SHA-256 of the SQL string (each unique string parsed once).
    func (c *SQLChecker) MustContainOrgFilter(sql string) error {
        key := hashSQL(sql)

        c.mu.RLock()
        if cached, ok := c.cache[key]; ok {
            c.mu.RUnlock()
            return cached
        }
        c.mu.RUnlock()

        result := c.classify(sql)

        c.mu.Lock()
        c.cache[key] = result
        c.mu.Unlock()
        return result
    }

    // classify parses the SQL and decides whether it references org_id legitimately.
    func (c *SQLChecker) classify(sql string) error {
        tree, err := pg_query.Parse(sql)
        if err != nil {
            // Unparseable SQL — fail safe: reject. Validator panic surfaces it.
            return fmt.Errorf("orgdb: cannot parse sql: %w", err)
        }
        if len(tree.Stmts) == 0 {
            return ErrSQLMissingOrgFilter
        }
        for _, raw := range tree.Stmts {
            stmt := raw.Stmt
            switch n := stmt.Node.(type) {
            case *pg_query.Node_SelectStmt, *pg_query.Node_UpdateStmt, *pg_query.Node_DeleteStmt:
                if !containsOrgIDColumnRef(stmt) {
                    return ErrSQLMissingOrgFilter
                }
            case *pg_query.Node_InsertStmt:
                if !insertHasOrgIDColumn(n.InsertStmt) {
                    return ErrSQLMissingOrgFilter
                }
            default:
                // DDL (CreateStmt, AlterTableStmt, DropStmt, IndexStmt, etc.):
                // these are migration-time. Accept; bypass path is the legitimate caller.
                // _ = n is a deliberate no-op to keep the type switch explicit.
                _ = n
            }
        }
        return nil
    }

    // containsOrgIDColumnRef walks the parse tree looking for any ColumnRef
    // whose name equals "org_id" used in a comparison or IN expression.
    // Recurses into nested WhereClause, JoinExpr.Quals, BoolExpr children,
    // SelectStmt subqueries — the common shapes Phase 1 SQL exercises.
    func containsOrgIDColumnRef(n *pg_query.Node) bool {
        if n == nil {
            return false
        }
        // Use the pg_query_go AST walker via marshalling to JSON + scanning, OR
        // recursively descend using a manual switch. For Phase 1 the simplest
        // correct check (per Pitfall 2 final paragraph) is: serialize the parse
        // tree to its protobuf text representation and search for a ColumnRef
        // whose fields contain a string node "org_id". This is a robust shortcut
        // that handles JOIN ... ON foo.org_id = bar.org_id without writing a
        // bespoke visitor for every Node oneof variant.
        text := n.String()
        // Look for the exact tokens pg_query emits for a ColumnRef on org_id.
        // pg_query's protobuf text format uses field labels; column refs look
        // like: column_ref:{fields:{string:{sval:"org_id"}} ... }
        return strings.Contains(text, `sval:"org_id"`)
    }

    // insertHasOrgIDColumn checks INSERT statements have org_id in the column list.
    func insertHasOrgIDColumn(ins *pg_query.InsertStmt) bool {
        if ins == nil {
            return false
        }
        for _, col := range ins.Cols {
            rt, ok := col.Node.(*pg_query.Node_ResTarget)
            if !ok {
                continue
            }
            if rt.ResTarget != nil && rt.ResTarget.Name == "org_id" {
                return true
            }
        }
        return false
    }

    func hashSQL(sql string) string {
        sum := sha256.Sum256([]byte(sql))
        return hex.EncodeToString(sum[:])
    }
    ```
    Implementation notes that the executor MUST verify against pg_query_go v6 API:
    - `pg_query.Parse(sql)` returns `*pg_query.ParseResult` with `Stmts []*RawStmt`; each `RawStmt.Stmt *Node`. Adjust field access if v6 differs.
    - The "search for `sval:"org_id"` in the protobuf text" approach is a pragmatic Phase 1 shortcut. If pg_query_go's `proto.Marshal` text format does not include that literal, fall back to a recursive walker that calls `pg_query.Walk` (or `pg_query.Deparse` and grep). The acceptance criterion is: ALL test cases in sqlcheck_test.go pass. If the walker shortcut fails, write a real visitor — Pitfall 2 mandates correctness over brevity.
    - DDL acceptance is correct because migrations go through `WithBypass` (D-11). The validator never sees DDL in production code paths.

    ```go
    // preflight runs the three checks in order:
    //   1. bypass marker present -> slog.Warn, skip remaining checks (D-04)
    //   2. ctx carries org_id (D-03)
    //   3. sql references org_id (D-02) -> cached by SHA-256 (sqlcheck.go)
    //
    // On failure in step 3: panic in ValidationPanic mode (dev/test), return error
    // in ValidationError mode (prod).
    func (o *OrgDB) preflight(ctx context.Context, sql string) error {
        // 1. Bypass shortcut (D-04, D-05).
        if reason, ok := BypassReason(ctx); ok {
            caller, _ := BypassCaller(ctx)
            slog.WarnContext(ctx, "orgdb bypass",
                "event", "orgdb_bypass",
                "reason", reason,
                "caller", caller,
                "sql_hash", hashSQL(sql),
            )
            return nil
        }

        // 2. ctx must carry org_id (D-03). Note: org_id presence does NOT mean it's
        // forwarded to the query — that's step 3's job. This step proves the
        // request has been through OrgContext middleware.
        if _, ok := orgkey.OrgIDFromContext(ctx); !ok {
            return ErrOrgIDMissingFromContext
        }

        // 3. SQL must reference org_id (D-02).
        if err := o.checker.MustContainOrgFilter(sql); err != nil {
            if o.mode == ValidationPanic {
                panic(err)
            }
            return err
        }
        return nil
    }
    ```

    ```go
    func main() {
        slog.SetDefault(slog.New(slog.NewJSONHandler(os.Stdout, &slog.HandlerOptions{Level: slog.LevelInfo})))

        cfg := config.MustLoad()

        sqlDB, err := sql.Open("pgx", cfg.DatabaseURL)
        if err != nil {
            slog.Error("migrate: open db", "err", err)
            os.Exit(1)
        }
        defer sqlDB.Close()

        if err := sqlDB.Ping(); err != nil {
            slog.Error("migrate: ping db", "err", err)
            os.Exit(1)
        }

        driver, err := pgmigrate.WithInstance(sqlDB, &pgmigrate.Config{})
        if err != nil {
            slog.Error("migrate: build driver", "err", err)
            os.Exit(1)
        }

        m, err := migrate.NewWithDatabaseInstance(
            "file://migrations", // expects working dir == repo root (Taskfile's `task migrate-up` runs from there)
            "postgres",
            driver,
        )
        if err != nil {
            slog.Error("migrate: new migrate", "err", err)
            os.Exit(1)
        }

        // D-11: emit the bypass event for audit even though golang-migrate does
        // not flow through orgDB. The bypass marker on ctx is documentary in
        // Phase 1 (no orgDB.Exec calls happen here) but proves the audit trail.
        ctx := db.WithBypass(context.Background(), "schema_migration")
        slog.WarnContext(ctx, "orgdb bypass",
            "event", "orgdb_bypass",
            "reason", "schema_migration",
            "caller", "cmd/migrate",
        )

        if err := m.Up(); err != nil && !errors.Is(err, migrate.ErrNoChange) {
            slog.Error("migrate: up", "err", err)
            os.Exit(1)
        }

        ver, dirty, verr := m.Version()
        if verr != nil && !errors.Is(verr, migrate.ErrNilVersion) {
            slog.Error("migrate: version", "err", verr)
            os.Exit(1)
        }
        slog.Info("migrations applied", "version", ver, "dirty", dirty)
    }
    ```
