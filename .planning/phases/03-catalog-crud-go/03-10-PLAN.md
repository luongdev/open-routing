// GetReadyz — readiness probe; pings DB pool + Redis. Returns 200 only when
// both back-ends respond; 500 otherwise. Bypass routing means no X-Org-Id;
// orgDB is not consulted here (orgDB rejects unscoped queries — calling
// Pool.Ping directly avoids the SQLChecker entirely).
func (h *Handlers) GetReadyz(ctx context.Context, req api.GetReadyzRequestObject) (api.GetReadyzResponseObject, error) {
    if err := h.deps.Pool.Ping(ctx); err != nil {
        h.deps.Logger.WarnContext(ctx, "readyz pool ping failed", "err", err)
        return api.GetReadyz500JSONResponse{InternalServerErrorJSONResponse: api.InternalServerErrorJSONResponse{
            Error: api.ErrorCodeInternal, Reason: "db_unavailable",
        }}, nil
    }
    if h.deps.Cache != nil {
        if err := h.deps.Cache.Ping(ctx); err != nil {
            h.deps.Logger.WarnContext(ctx, "readyz cache ping failed", "err", err)
            return api.GetReadyz500JSONResponse{InternalServerErrorJSONResponse: api.InternalServerErrorJSONResponse{
                Error: api.ErrorCodeInternal, Reason: "cache_unavailable",
            }}, nil
        }
    }
    return api.GetReadyz200JSONResponse{Status: "ready"}, nil
}

// postEntity is a helper: marshals body to JSON, sends POST with X-Org-Id;
// returns response code + decoded id field. Used by cross-org tests below.
func postEntity(t *testing.T, urlBase, entityPath string, orgID uuid.UUID, body any) (int, uuid.UUID) {
    t.Helper()
    b, err := json.Marshal(body)
    require.NoError(t, err)
    req, err := http.NewRequest(http.MethodPost,
        fmt.Sprintf("%s/v1/orgs/%s/%s", urlBase, orgID, entityPath),
        bytes.NewReader(b))
    require.NoError(t, err)
    req.Header.Set("X-Org-Id", orgID.String())
    req.Header.Set("Content-Type", "application/json")
    resp, err := http.DefaultClient.Do(req)
    require.NoError(t, err)
    defer resp.Body.Close()
    var out struct{ Id string `json:"id"` }
    _ = json.NewDecoder(resp.Body).Decode(&out)
    var id uuid.UUID
    if out.Id != "" {
        id = uuid.MustParse(out.Id)
    }
    return resp.StatusCode, id
}

// getEntityStatus returns the status code from GET {urlBase}/v1/orgs/{orgID}/{entityPath}/{id}.
func getEntityStatus(t *testing.T, urlBase, entityPath string, orgID, id uuid.UUID) int {
    t.Helper()
    req, err := http.NewRequest(http.MethodGet,
        fmt.Sprintf("%s/v1/orgs/%s/%s/%s", urlBase, orgID, entityPath, id),
        nil)
    require.NoError(t, err)
    req.Header.Set("X-Org-Id", orgID.String())
    resp, err := http.DefaultClient.Do(req)
    require.NoError(t, err)
    defer resp.Body.Close()
    return resp.StatusCode
}
