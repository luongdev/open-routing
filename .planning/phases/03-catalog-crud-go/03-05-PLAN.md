// notimpl.go until those phases land — see D-69 for the carry-forward
// strategy via struct embedding (D-70).
//

// break_reasons,agent_skills}.go (D-68). Shared helpers live in mappers.go,
// cursor.go, errors.go.
package catalog

import (
    "context"
    "log/slog"
    "net/http"
    "time"

    "github.com/jackc/pgx/v5/pgxpool"

    "github.com/luongdev/open-routing/services/api/internal/api"
    "github.com/luongdev/open-routing/services/api/internal/cache"
    "github.com/luongdev/open-routing/services/api/internal/db"
)

// Deps bundles required runtime deps for the catalog handler set. Every
// field is REQUIRED; a nil field panics at first call. D-71 hybrid
// constructor pattern: required fields in this struct, optional knobs via
// Option func.
type Deps struct {
    OrgDB  *db.OrgDB
    Pool   *pgxpool.Pool
    Cache  *cache.Cache
    Logger *slog.Logger
}

// Option mutates a *Handlers post-construction. Only WithClock is
// exposed in v0.1 (D-71 — for deterministic timestamps in tests).
type Option func(*Handlers)

// WithClock overrides the clock used for created_at/updated_at minting
// and UUIDv7 timestamps in tests. Default is time.Now.
func WithClock(c func() time.Time) Option {
    return func(h *Handlers) { h.clock = c }
}

// Handlers implements api.StrictServerInterface for every catalog
// endpoint. Per-method bodies live in entity-specific files; this file
// holds the constructor and shared state.
type Handlers struct {
    deps  Deps
    clock func() time.Time
}

// Compile-time guarantee that Handlers satisfies the generated interface.
// If this line fails to compile, a method is missing from the package —
// look at the diff between server.gen.go's StrictServerInterface and
// catalog's exported methods.
var _ api.StrictServerInterface = (*Handlers)(nil)

// New constructs a Handlers value. deps fields MUST be non-nil; passing
// a zero-value Deps will panic on the first call that dereferences a
// required field. Optional knobs apply via Option funcs (D-71).
func New(deps Deps, opts ...Option) *Handlers {
    h := &Handlers{deps: deps, clock: time.Now}
    for _, opt := range opts {
        opt(h)
    }
    return h
}

// notImplemented is used by the Wave 0 skeleton to satisfy the strict-
// server interface without writing real handlers. Plans 03-06..03-09
// REPLACE the body of each per-entity method with the real implementation;
// notImplemented is only called for Phase 4/5 endpoints (see notimpl.go).
//
// Returns a wrapped 500 response so the strict-server pipeline emits a
// well-formed error envelope with request_id. NOT a real 501 because the
// generated wrappers do not include 501 variants on every endpoint — using
// the 500 wrapper that every operation already has keeps the type switch
// in server.go::injectRequestIDIntoErrorResponse complete.
func notImplemented(opName string) error {
    return &notImplError{op: opName}
}

type notImplError struct{ op string }

func (e *notImplError) Error() string {
    return "catalog: " + e.op + " not implemented yet"
}

package catalog

import (
    "encoding/base64"
    "encoding/json"
    "errors"
    "time"

    "github.com/google/uuid"
)

// errBadCursor signals an undecodable / malformed cursor — callers map
// to 400 invalid_body.
var errBadCursor = errors.New("catalog: bad cursor")

// Cursor is the decoded payload of an opaque base64 cursor. Field
// shape matches D-63 verbatim — RFC3339Nano + UUIDv7. The composite
// (CreatedAt, ID) is stable under inserts because UUIDv7 carries time.
type Cursor struct {
    CreatedAt time.Time `json:"created_at"`
    ID        uuid.UUID `json:"id"`
}

// EncodeCursor packs (createdAt, id) into base64(JSON). The result is
// opaque to clients but server-decodable.
func EncodeCursor(createdAt time.Time, id uuid.UUID) (string, error) {
    b, err := json.Marshal(Cursor{CreatedAt: createdAt, ID: id})
    if err != nil {
        return "", err
    }
    return base64.StdEncoding.EncodeToString(b), nil
}

// DecodeCursor inverts EncodeCursor. The empty string returns (nil, nil)
// — the first-page case where no cursor is supplied. Any decode failure
// returns errBadCursor.
func DecodeCursor(s string) (*Cursor, error) {
    if s == "" {
        return nil, nil
    }
    b, err := base64.StdEncoding.DecodeString(s)
    if err != nil {
        return nil, errBadCursor
    }
    var c Cursor
    if err := json.Unmarshal(b, &c); err != nil {
        return nil, errBadCursor
    }
    return &c, nil
}

package catalog

import (
    "time"

    "github.com/google/uuid"
    "github.com/jackc/pgx/v5/pgtype"
)

// pgUUID converts a google/uuid value to pgtype.UUID for sqlc params.
func pgUUID(id uuid.UUID) pgtype.UUID {
    return pgtype.UUID{Bytes: id, Valid: true}
}

// pgUUIDFromBytes wraps a 16-byte array.
func pgUUIDFromBytes(b [16]byte) pgtype.UUID {
    return pgtype.UUID{Bytes: b, Valid: true}
}

// apiUUID converts a pgtype.UUID (from sqlc rows) to a google/uuid value.
// The Valid flag is NOT checked — sqlc emits Valid=true for non-null
// columns and the catalog schema has NO nullable UUID PRIMARY KEYs.
func apiUUID(p pgtype.UUID) uuid.UUID {
    return uuid.UUID(p.Bytes)
}

// pgTimestamptzPtr wraps an optional time pointer in pgtype.Timestamptz.
// Nil → Valid=false (NULL in SQL). Used for cursor params (sqlc.narg).
func pgTimestamptzPtr(t *time.Time) pgtype.Timestamptz {
    if t == nil {
        return pgtype.Timestamptz{Valid: false}
    }
    return pgtype.Timestamptz{Time: *t, Valid: true}
}

// pgTextPtr wraps an optional string pointer in pgtype.Text. Used for
// nullable name filter params.
func pgTextPtr(s *string) pgtype.Text {
    if s == nil {
        return pgtype.Text{Valid: false}
    }
    return pgtype.Text{String: *s, Valid: true}
}

// deref returns *p or the zero value of T when p is nil. Used to flatten
// UpdateXRequest pointer fields to concrete values for sqlc params.
func deref[T any](p *T) T {
    var zero T
    if p == nil {
        return zero
    }
    return *p
}

// derefOr returns *p or fallback when p is nil. Used for default page
// size (25) when the client omits the limit parameter.
func derefOr[T any](p *T, fallback T) T {
    if p == nil {
        return fallback
    }
    return *p
}

package catalog

import (
    "errors"
    "net/http"

    "github.com/jackc/pgx/v5"
    "github.com/jackc/pgx/v5/pgconn"

    "github.com/luongdev/open-routing/services/api/internal/api"
)

// mapPgError translates common pgx / pgconn errors to the locked
// (http status, ErrorCode, reason) triple. Catalog handlers use this for
// every non-happy-path branch:
//
//   - pgx.ErrNoRows → 404 / not_found / "{entity}_not_found"
//   - pgconn.PgError{Code: "23505" UNIQUE violation} → 409 / version_conflict / "external_id_collision"
//   - anything else → 500 / internal / "internal"
//
// The handler then wraps the triple in the appropriate generated
// *JSONResponse type. mapPgError does NOT know about the response type;
// it returns only the status+code+reason.
func mapPgError(err error, entity string) (int, api.ErrorCode, string) {
    if errors.Is(err, pgx.ErrNoRows) {
        return http.StatusNotFound, api.ErrorCodeNotFound, entity + "_not_found"
    }
    var pgErr *pgconn.PgError
    if errors.As(err, &pgErr) && pgErr.Code == "23505" {
        return http.StatusConflict, api.ErrorCodeVersionConflict, "external_id_collision"
    }
    return http.StatusInternalServerError, api.ErrorCodeInternal, "internal"
}

func TestCursorRoundTrip(t *testing.T) {
    createdAt := time.Date(2026, 5, 16, 9, 30, 45, 123456789, time.UTC)
    id := uuid.Must(uuid.NewV7())
    s, err := EncodeCursor(createdAt, id)
    require.NoError(t, err)
    require.NotEmpty(t, s)
    cur, err := DecodeCursor(s)
    require.NoError(t, err)
    require.NotNil(t, cur)
    require.Equal(t, id, cur.ID)
    require.WithinDuration(t, createdAt, cur.CreatedAt, time.Microsecond)
}

func TestDecodeCursor_Empty(t *testing.T) {
    cur, err := DecodeCursor("")
    require.NoError(t, err)
    require.Nil(t, cur)
}

func TestDecodeCursor_Malformed(t *testing.T) {
    _, err := DecodeCursor("not-base64!!!")
    require.ErrorIs(t, err, errBadCursor)
}

func TestDecodeCursor_BadJSON(t *testing.T) {
    // valid base64 but not valid JSON
    s := "bm90LWpzb24=" // base64("not-json")
    _, err := DecodeCursor(s)
    require.ErrorIs(t, err, errBadCursor)
}

func TestEncodeCursor_Stable(t *testing.T) {
    // Same input → same output (no randomness)
    createdAt := time.Date(2026, 5, 16, 9, 30, 45, 0, time.UTC)
    id := uuid.MustParse("01900000-0000-7000-8000-000000000001")
    s1, _ := EncodeCursor(createdAt, id)
    s2, _ := EncodeCursor(createdAt, id)
    require.Equal(t, s1, s2)
}
