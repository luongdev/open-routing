**`services/api/internal/cache/cache.go` target shape:**

```go
// Package cache implements the v0.1 Redis-backed read-through cache for
// single-entity GETs in the catalog (CAT-11). It is intentionally
// entity-agnostic: callers pass typed loaders via `GetOrSet[T any]` and
// receive typed values back.
//
// Locked contracts (CONTEXT.md D-49..D-60):
//   - Cache contains DTO values, NOT sqlc rows (D-49).
//   - One shared *Cache for all entity types — generics, not per-entity instances (D-50/D-51).
//   - Miss path dedup'd via singleflight per process (D-52).
//   - Refresh-ahead fires when PTTL < 10s, async, via context.WithoutCancel (D-53).
//   - No negative caching — loader returning ErrNotFound is NOT cached (D-54).
//   - Del is best-effort; callers MUST log+continue on failure (D-55).
//   - Caller passes ttl; locked at 60s by D-59. Helper does not constant the value.
//   - JSON via encoding/json stdlib (D-60).
//   - Observability via slog only; no OTel metrics (D-57).
package cache

import (
    "context"
    "encoding/json"
    "errors"
    "fmt"
    "log/slog"
    "time"

    "github.com/redis/go-redis/v9"
    "golang.org/x/sync/singleflight"
)

// ErrNotFound is the sentinel a loader returns to signal "row absent" so
// GetOrSet can propagate without writing an empty cache entry (D-54).
var ErrNotFound = errors.New("cache: not found")

// refreshThreshold is the locked 10s remaining window per D-53.
const refreshThreshold = 10 * time.Second

// Cache wraps a Redis client + singleflight group + slog logger. Construct
// once at boot via New(rdb, logger); pass into every consumer via Deps.
type Cache struct {
    rdb    *redis.Client
    sf     *singleflight.Group
    logger *slog.Logger
}

// New constructs a Cache. rdb and logger MUST be non-nil; passing nil logger
// will panic on first call when slog dereferences it.
func New(rdb *redis.Client, logger *slog.Logger) *Cache {
    return &Cache{rdb: rdb, sf: &singleflight.Group{}, logger: logger}
}

// Key returns the canonical cache key "or:{orgID}:{entity}:{id}" per D-58.
// Helper constructor — every catalog handler calls this rather than building
// the string manually.
func Key(orgID, entity, id any) string {
    return fmt.Sprintf("or:%s:%s:%s", orgID, entity, id)
}

// GetOrSet looks up `key` in Redis; on hit unmarshals JSON to T and returns;
// on miss invokes `load`, marshals the result, writes with `ttl`, and
// returns. Concurrent misses for the same `key` are deduped via singleflight
// (D-52). On hit, if remaining TTL < 10s, spawns an async refresh goroutine
// using `context.WithoutCancel(ctx)` so the request finishing does NOT cancel
// the refresh (D-53). If `load` returns ErrNotFound, propagates without
// caching the empty value (D-54).
func GetOrSet[T any](
    ctx context.Context,
    c *Cache,
    key string,
    ttl time.Duration,
    load func(context.Context) (T, error),
) (T, error) {
    var zero T

    // (1) Hit path
    raw, err := c.rdb.Get(ctx, key).Bytes()
    if err == nil {
        var v T
        if jerr := json.Unmarshal(raw, &v); jerr == nil {
            c.logger.DebugContext(ctx, "cache", "key", key, "outcome", "hit")
            // refresh-ahead check (D-53)
            if pttl, perr := c.rdb.PTTL(ctx, key).Result(); perr == nil && pttl > 0 && pttl < refreshThreshold {
                bgCtx := context.WithoutCancel(ctx)
                go func() {
                    _, _, _ = c.sf.Do("refresh:"+key, func() (any, error) {
                        c.logger.DebugContext(bgCtx, "cache", "key", key, "outcome", "refresh")
                        nv, lerr := load(bgCtx)
                        if lerr != nil {
                            c.logger.WarnContext(bgCtx, "cache refresh load", "key", key, "err", lerr)
                            return nil, lerr
                        }
                        nb, jerr := json.Marshal(nv)
                        if jerr != nil {
                            c.logger.WarnContext(bgCtx, "cache refresh marshal", "key", key, "err", jerr)
                            return nil, jerr
                        }
                        if serr := c.rdb.Set(bgCtx, key, nb, ttl).Err(); serr != nil {
                            c.logger.WarnContext(bgCtx, "cache refresh set", "key", key, "err", serr)
                        }
                        return nv, nil
                    })
                }()
            }
            return v, nil
        }
        // Unmarshal failed — fall through to loader; log at warn.
        c.logger.WarnContext(ctx, "cache unmarshal", "key", key, "err", jerr)
    } else if !errors.Is(err, redis.Nil) {
        // Redis error (not "missing key") — log; continue to loader so a
        // Redis outage does not turn into a 5xx (D-55 spirit).
        c.logger.WarnContext(ctx, "cache get", "key", key, "err", err)
    }

    // (2) Miss path: singleflight-dedup'd load
    res, sferr, _ := c.sf.Do(key, func() (any, error) {
        c.logger.DebugContext(ctx, "cache", "key", key, "outcome", "miss")
        nv, lerr := load(ctx)
        if lerr != nil {
            // Surface ErrNotFound so caller maps to 404 — do NOT write empty value (D-54).
            return nv, lerr
        }
        nb, jerr := json.Marshal(nv)
        if jerr != nil {
            c.logger.WarnContext(ctx, "cache marshal", "key", key, "err", jerr)
            return nv, jerr
        }
        if serr := c.rdb.Set(ctx, key, nb, ttl).Err(); serr != nil {
            c.logger.WarnContext(ctx, "cache set", "key", key, "err", serr)
            // Continue — return value to caller anyway.
        }
        return nv, nil
    })
    if sferr != nil {
        if errors.Is(sferr, ErrNotFound) {
            return zero, ErrNotFound
        }
        return zero, sferr
    }
    typed, ok := res.(T)
    if !ok {
        // Should be unreachable — singleflight result is whatever load returned.
        return zero, fmt.Errorf("cache: singleflight produced unexpected type %T", res)
    }
    return typed, nil
}

// Del removes a cache entry. Returns the underlying Redis error; callers
// MUST log+continue on failure per D-55 — a cache deletion failure NEVER
// converts a successful DB write into a 5xx.
func (c *Cache) Del(ctx context.Context, key string) error {
    if err := c.rdb.Del(ctx, key).Err(); err != nil {
        c.logger.WarnContext(ctx, "cache", "key", key, "outcome", "error", "op", "del", "err", err)
        return err
    }
    c.logger.DebugContext(ctx, "cache", "key", key, "outcome", "del")
    return nil
}
```

package cache_test

import (
    "context"
    "errors"
    "io"
    "log/slog"
    "sync"
    "sync/atomic"
    "testing"
    "time"

    "github.com/alicebob/miniredis/v2"
    "github.com/redis/go-redis/v9"
    "github.com/stretchr/testify/require"

    "github.com/luongdev/open-routing/services/api/internal/cache"
)

type fakeAgent struct {
    ID   string `json:"id"`
    Name string `json:"name"`
}

func newCache(t *testing.T) (*cache.Cache, *miniredis.Miniredis) {
    t.Helper()
    s := miniredis.RunT(t)
    rdb := redis.NewClient(&redis.Options{Addr: s.Addr()})
    t.Cleanup(func() { _ = rdb.Close() })
    return cache.New(rdb, slog.New(slog.NewTextHandler(io.Discard, nil))), s
}

func TestGetOrSet_Miss_LoadsAndSets(t *testing.T) {
    c, s := newCache(t)
    ctx := context.Background()
    key := cache.Key("orgA", "agents", "ag-1")
    loads := atomic.Int32{}
    got, err := cache.GetOrSet[fakeAgent](ctx, c, key, 60*time.Second,
        func(ctx context.Context) (fakeAgent, error) {
            loads.Add(1)
            return fakeAgent{ID: "ag-1", Name: "alice"}, nil
        })
    require.NoError(t, err)
    require.Equal(t, "alice", got.Name)
    require.Equal(t, int32(1), loads.Load())
    require.True(t, s.Exists(key), "miss path MUST write to redis")
}

func TestGetOrSet_Hit_NoLoad(t *testing.T) {
    c, s := newCache(t)
    ctx := context.Background()
    key := cache.Key("orgA", "agents", "ag-1")

    // Pre-warm the cache.
    _, err := cache.GetOrSet[fakeAgent](ctx, c, key, 60*time.Second,
        func(ctx context.Context) (fakeAgent, error) {
            return fakeAgent{ID: "ag-1", Name: "alice"}, nil
        })
    require.NoError(t, err)

    // Second call must NOT invoke loader.
    loads := atomic.Int32{}
    got, err := cache.GetOrSet[fakeAgent](ctx, c, key, 60*time.Second,
        func(ctx context.Context) (fakeAgent, error) {
            loads.Add(1)
            return fakeAgent{ID: "ag-1", Name: "different"}, nil
        })
    require.NoError(t, err)
    require.Equal(t, "alice", got.Name, "hit MUST return cached value")
    require.Equal(t, int32(0), loads.Load(), "hit MUST NOT invoke loader")
    _ = s
}

func TestGetOrSet_ErrNotFound_NotCached(t *testing.T) {
    c, s := newCache(t)
    ctx := context.Background()
    key := cache.Key("orgA", "agents", "missing")
    _, err := cache.GetOrSet[fakeAgent](ctx, c, key, 60*time.Second,
        func(ctx context.Context) (fakeAgent, error) {
            return fakeAgent{}, cache.ErrNotFound
        })
    require.ErrorIs(t, err, cache.ErrNotFound)
    require.False(t, s.Exists(key), "ErrNotFound MUST NOT write to redis (D-54)")
}

func TestGetOrSet_RefreshAhead(t *testing.T) {
    c, s := newCache(t)
    ctx := context.Background()
    key := cache.Key("orgA", "agents", "ag-1")
    loads := atomic.Int32{}
    loader := func(ctx context.Context) (fakeAgent, error) {
        loads.Add(1)
        return fakeAgent{ID: "ag-1", Name: "v" + string(rune('0'+loads.Load()))}, nil
    }
    // Initial miss + set with 60s TTL.
    _, err := cache.GetOrSet[fakeAgent](ctx, c, key, 60*time.Second, loader)
    require.NoError(t, err)
    require.Equal(t, int32(1), loads.Load())

    // Fast-forward miniredis clock so PTTL < 10s.
    s.FastForward(51 * time.Second)

    // Next hit should trigger async refresh — loader fires again.
    _, err = cache.GetOrSet[fakeAgent](ctx, c, key, 60*time.Second, loader)
    require.NoError(t, err)

    // Wait briefly for the async goroutine to land.
    require.Eventually(t, func() bool {
        return loads.Load() >= 2
    }, 2*time.Second, 10*time.Millisecond, "refresh-ahead goroutine MUST fire when PTTL < 10s")
}

func TestGetOrSet_Singleflight_Dedups(t *testing.T) {
    c, _ := newCache(t)
    ctx := context.Background()
    key := cache.Key("orgA", "agents", "ag-x")
    loads := atomic.Int32{}
    loader := func(ctx context.Context) (fakeAgent, error) {
        loads.Add(1)
        time.Sleep(20 * time.Millisecond) // hold the singleflight so concurrent callers join
        return fakeAgent{ID: "ag-x", Name: "alice"}, nil
    }

    var wg sync.WaitGroup
    const N = 32
    for i := 0; i < N; i++ {
        wg.Add(1)
        go func() {
            defer wg.Done()
            _, err := cache.GetOrSet[fakeAgent](ctx, c, key, 60*time.Second, loader)
            require.NoError(t, err)
        }()
    }
    wg.Wait()
    require.Equal(t, int32(1), loads.Load(), "singleflight MUST dedup concurrent misses to ONE load (D-52)")
}

func TestDel(t *testing.T) {
    c, s := newCache(t)
    ctx := context.Background()
    key := cache.Key("orgA", "agents", "ag-1")
    _, _ = cache.GetOrSet[fakeAgent](ctx, c, key, 60*time.Second,
        func(ctx context.Context) (fakeAgent, error) {
            return fakeAgent{ID: "ag-1", Name: "alice"}, nil
        })
    require.True(t, s.Exists(key))
    require.NoError(t, c.Del(ctx, key))
    require.False(t, s.Exists(key), "Del MUST remove the key")
}

func TestKey_Format(t *testing.T) {
    require.Equal(t, "or:orgA:agents:ag-1", cache.Key("orgA", "agents", "ag-1"))
    require.Equal(t, "or:01900000-0000-7000-8000-000000000000:channels:c-1",
        cache.Key("01900000-0000-7000-8000-000000000000", "channels", "c-1"))
}

func TestDel_RedisErrorPropagates(t *testing.T) {
    c, s := newCache(t)
    ctx := context.Background()
    s.Close() // simulate Redis outage
    err := c.Del(ctx, "or:any:key")
    require.Error(t, err, "Del MUST propagate redis errors (caller log+continues per D-55)")
}

Hazards to avoid:
- **Pitfall 4**: refresh goroutine MUST use `context.WithoutCancel(ctx)`, NOT `context.Background()`. The former preserves trace_id + org_id; the latter strips them and would panic orgDB in dev when the refresh tries to query.
- **D-54 regression**: do not call `rdb.Set` when loader returns `ErrNotFound`. The test `TestGetOrSet_ErrNotFound_NotCached` enforces this via `require.False(t, s.Exists(key), ...)`.
