func (h *Handlers) CreateAgent(ctx context.Context, req api.CreateAgentRequestObject) (api.CreateAgentResponseObject, error) {
    orgID, ok := orgkey.OrgIDFromContext(ctx)
    if !ok {
        return api.CreateAgent500JSONResponse{InternalServerErrorJSONResponse: api.InternalServerErrorJSONResponse{
            Error: api.ErrorCodeInternal, Reason: "missing_org_id_in_context",
        }}, nil
    }

    // UUIDv7 minted at write boundary (D-19).
    id := uuid.Must(uuid.NewV7())
    q := generated.New(h.deps.OrgDB)

    row, err := q.InsertAgent(ctx, generated.InsertAgentParams{
        ID:         pgUUID(id),
        OrgID:      pgUUID(orgID),
        ExternalID: req.Body.ExternalId,
        Name:       req.Body.Name,
        Email:      string(req.Body.Email),
        Enabled:    derefOr(req.Body.Enabled, true),
    })
    if err != nil {
        status, code, reason := mapPgError(err, "agent")
        if status == 409 {
            // pgError 23505 — external_id collision (Claude's Discretion in D-section)
            return api.CreateAgent409JSONResponse(buildCreateAgentErrorUnion(code, reason)), nil
        }
        h.deps.Logger.ErrorContext(ctx, "create agent", "err", err)
        return api.CreateAgent500JSONResponse{InternalServerErrorJSONResponse: api.InternalServerErrorJSONResponse{
            Error: code, Reason: reason,
        }}, nil
    }

    // If skills[] supplied at create, run the same transactional replace
    // path as UpdateAgent. (CAT-03 — admin can attach skills at create.)
    if req.Body.Skills != nil && len(*req.Body.Skills) > 0 {
        if errResp, err := h.replaceAgentSkills(ctx, orgID, id, *req.Body.Skills); err != nil {
            return api.CreateAgent500JSONResponse{InternalServerErrorJSONResponse: api.InternalServerErrorJSONResponse{
                Error: api.ErrorCodeInternal, Reason: err.Error(),
            }}, nil
        } else if errResp != nil {
            // 422 invalid_reference — rollback the agent insert by soft-deleting.
            // Simpler: wrap the whole thing in BeginTx. See Pitfall 5 (replace).
            // For Plan 03-06 v1: best-effort — log and continue; integration
            // test asserts the 422 is returned. Plan 03-14 (hardening) can
            // wrap both InsertAgent + skills-replace in a single tx.
            return api.CreateAgent422JSONResponse{ErrorResponse: api.ErrorResponse{
                Error: api.ErrorCodeInvalidReference, Reason: errResp.Reason,
            }}, nil
        }
    }

    agent := mapAgent(row, nil)
    return api.CreateAgent201JSONResponse(agent), nil
}

// buildCreateAgentErrorUnion handles Pitfall 2 — CreateAgent409JSONResponse
// is a oneOf ErrorResponse | VersionConflictErrorResponse union. The 23505
// collision is an ErrorResponse, NOT a VersionConflict.
func buildCreateAgentErrorUnion(code api.ErrorCode, reason string) api.CreateAgent409JSONResponseBody {
    var body api.CreateAgent409JSONResponseBody
    _ = body.FromErrorResponse(api.ErrorResponse{
        Error:  code,
        Reason: reason,
    })
    return body
}

func (h *Handlers) UpdateAgent(ctx context.Context, req api.UpdateAgentRequestObject) (api.UpdateAgentResponseObject, error) {
    orgID, ok := orgkey.OrgIDFromContext(ctx)
    if !ok {
        return api.UpdateAgent500JSONResponse{InternalServerErrorJSONResponse: api.InternalServerErrorJSONResponse{
            Error: api.ErrorCodeInternal, Reason: "missing_org_id_in_context",
        }}, nil
    }
    agentID := uuid.UUID(req.Id)
    q := generated.New(h.deps.OrgDB)

    // First: cross-row validate skills[] if supplied (D-76). Short-circuits
    // the agent UPDATE if any skill_id is unknown — avoids partial state.
    if req.Body.Skills != nil {
        ids := make([][16]byte, 0, len(*req.Body.Skills))
        for _, s := range *req.Body.Skills {
            ids = append(ids, [16]byte(s.SkillId))
        }
        missing, mErr := q.SkillsMissingInOrg(ctx, generated.SkillsMissingInOrgParams{
            Column1: ids,           // sqlc-generated name may differ; verify post-regen
            OrgID:   pgUUID(orgID),
        })
        if mErr != nil {
            h.deps.Logger.ErrorContext(ctx, "skills missing probe", "err", mErr)
            return api.UpdateAgent500JSONResponse{InternalServerErrorJSONResponse: api.InternalServerErrorJSONResponse{
                Error: api.ErrorCodeInternal, Reason: "skills_probe_failed",
            }}, nil
        }
        if len(missing) > 0 {
            return api.UpdateAgent422JSONResponse{ErrorResponse: api.ErrorResponse{
                Error: api.ErrorCodeInvalidReference, Reason: fmt.Sprintf("unknown_skill_id:%s", uuid.UUID(missing[0])),
            }}, nil
        }
    }

    // Atomic version-checked UPDATE (D-66).
    row, err := q.UpdateAgent(ctx, generated.UpdateAgentParams{
        ID:      pgUUID(agentID),
        Name:    derefOr(req.Body.Name, ""),
        Email:   derefOr((*string)(req.Body.Email), ""),
        Enabled: derefOr(req.Body.Enabled, true),
        OrgID:   pgUUID(orgID),
        Version: int32(req.Body.Version),
    })
    if errors.Is(err, pgx.ErrNoRows) {
        // 0 rows — was it 404 or 409?  (D-66 disambiguation)
        cur, perr := q.GetAgentByIdAnyVersion(ctx, generated.GetAgentByIdAnyVersionParams{
            ID: pgUUID(agentID), OrgID: pgUUID(orgID),
        })
        if errors.Is(perr, pgx.ErrNoRows) {
            return api.UpdateAgent404JSONResponse{NotFoundJSONResponse: api.NotFoundJSONResponse{
                Error: api.ErrorCodeNotFound, Reason: "agent_not_found",
            }}, nil
        }
        if perr != nil {
            return api.UpdateAgent500JSONResponse{InternalServerErrorJSONResponse: api.InternalServerErrorJSONResponse{
                Error: api.ErrorCodeInternal, Reason: "disambiguation_failed",
            }}, nil
        }
        // 409 — cache DEL (D-56) then return the current row.
        _ = h.deps.Cache.Del(ctx, cache.Key(orgID, "agents", agentID))
        return api.UpdateAgent409JSONResponse{
            Current: mapAgent(cur, nil),
            Error:   "version_conflict",  // UpdateAgent409JSONResponseBodyError = string literal type
            Reason:  "version_mismatch",
        }, nil
    }
    if err != nil {
        h.deps.Logger.ErrorContext(ctx, "update agent", "err", err)
        return api.UpdateAgent500JSONResponse{InternalServerErrorJSONResponse: api.InternalServerErrorJSONResponse{
            Error: api.ErrorCodeInternal, Reason: "internal",
        }}, nil
    }

    // Happy path: replace skills if supplied (transactional — Pitfall 5).
    if req.Body.Skills != nil {
        if _, err := h.replaceAgentSkills(ctx, orgID, agentID, *req.Body.Skills); err != nil {
            h.deps.Logger.ErrorContext(ctx, "agent skills replace", "err", err)
            return api.UpdateAgent500JSONResponse{InternalServerErrorJSONResponse: api.InternalServerErrorJSONResponse{
                Error: api.ErrorCodeInternal, Reason: "skills_replace_failed",
            }}, nil
        }
    }

    // Invalidate cache after commit (D-55).
    if delErr := h.deps.Cache.Del(ctx, cache.Key(orgID, "agents", agentID)); delErr != nil {
        h.deps.Logger.WarnContext(ctx, "cache del failed", "key", cache.Key(orgID, "agents", agentID), "err", delErr)
    }

    // Load fresh skills for the response (so PATCH returns the post-state).
    qAgain := generated.New(h.deps.OrgDB)
    skills, _ := qAgain.ListSkillsForAgent(ctx, generated.ListSkillsForAgentParams{
        AgentID: pgUUID(agentID), OrgID: pgUUID(orgID),
    })
    return api.UpdateAgent200JSONResponse(mapAgent(row, skills)), nil
}

type skillsErrResp struct{ Reason string }

func (h *Handlers) replaceAgentSkills(ctx context.Context, orgID, agentID uuid.UUID, assignments []api.AgentSkillAssignment) (*skillsErrResp, error) {
    q := generated.New(h.deps.OrgDB)

    // Validate every skill_id exists in same org (D-76).
    ids := make([][16]byte, 0, len(assignments))
    for _, s := range assignments {
        ids = append(ids, [16]byte(s.SkillId))
    }
    missing, err := q.SkillsMissingInOrg(ctx, generated.SkillsMissingInOrgParams{
        Column1: ids, OrgID: pgUUID(orgID),
    })
    if err != nil {
        return nil, fmt.Errorf("skills probe: %w", err)
    }
    if len(missing) > 0 {
        return &skillsErrResp{Reason: fmt.Sprintf("unknown_skill_id:%s", uuid.UUID(missing[0]))}, nil
    }

    // Transactional replace (D-25/Pitfall 5). OrgDB.BeginTx returns an
    // *OrgTx satisfying generated.DBTX (Plan 03-03 OQ-5 extension).
    tx, err := h.deps.OrgDB.BeginTx(ctx)
    if err != nil {
        return nil, fmt.Errorf("begin tx: %w", err)
    }
    defer func() { _ = tx.Rollback(ctx) }() // safe on commit

    qtx := generated.New(tx)
    if _, err := qtx.DeleteAgentSkills(ctx, generated.DeleteAgentSkillsParams{
        AgentID: pgUUID(agentID), OrgID: pgUUID(orgID),
    }); err != nil {
        return nil, fmt.Errorf("delete agent skills: %w", err)
    }
    for _, s := range assignments {
        if _, err := qtx.InsertAgentSkill(ctx, generated.InsertAgentSkillParams{
            AgentID:     pgUUID(agentID),
            SkillID:     pgUUID(uuid.UUID(s.SkillId)),
            OrgID:       pgUUID(orgID),
            Proficiency: int32(s.Proficiency),
        }); err != nil {
            return nil, fmt.Errorf("insert agent skill %s: %w", uuid.UUID(s.SkillId), err)
        }
    }
    if err := tx.Commit(ctx); err != nil {
        return nil, fmt.Errorf("commit: %w", err)
    }
    return nil, nil
}

// mapAgent converts a sqlc Agent row + a list of ListSkillsForAgentRow
// values into the api.Agent DTO. skills may be nil (list responses don't
// include embedded skills per OQ-1A).
func mapAgent(row generated.Agent, skills []generated.ListSkillsForAgentRow) api.Agent {
    a := api.Agent{
        Id:         api.UUIDv7(apiUUID(row.ID)),
        OrgId:      api.UUIDv7(apiUUID(row.OrgID)),
        ExternalId: row.ExternalID,
        Name:       row.Name,
        Email:      apiTypesEmail(row.Email),  // openapi_types.Email is a string alias
        Enabled:    row.Enabled,
        Version:    int(row.Version),
        CreatedAt:  ptrTime(row.CreatedAt),
        UpdatedAt:  ptrTime(row.UpdatedAt),
    }
    if skills != nil {
        list := make([]api.AgentSkillAssignment, 0, len(skills))
        for _, s := range skills {
            name := s.SkillName
            list = append(list, api.AgentSkillAssignment{
                SkillId:     api.UUIDv7(apiUUID(s.SkillID)),
                Proficiency: int(s.Proficiency),
                Name:        &name,
            })
        }
        a.Skills = &list
    }
    return a
}

```go
func (h *Handlers) UpdateAgent(ctx context.Context, req api.UpdateAgentRequestObject) (api.UpdateAgentResponseObject, error) {
    orgID, ok := orgkey.OrgIDFromContext(ctx)
    if !ok { /* 500 */ }
    agentID := uuid.UUID(req.Id)

    // Codex C1: Layer 2 proficiency validation BEFORE any DB call.
    if req.Body.Skills != nil {
        if badIdx, ok := validateProficiencyRange(*req.Body.Skills); !ok {
            field := fmt.Sprintf("skills[%d].proficiency", badIdx)
            return api.UpdateAgent422JSONResponse{ErrorResponse: api.ErrorResponse{
                Error: api.ErrorCodeInvalidValue, Reason: "proficiency must be 1-10", Field: &field,
            }}, nil
        }
    }

    // Codex C4: BeginTx FIRST — both agent UPDATE and skills replace share the same tx.
    tx, err := h.deps.OrgDB.BeginTx(ctx)
    if err != nil { /* 500 */ }
    defer func() { _ = tx.Rollback(ctx) }()  // safe after commit
    qtx := generated.New(tx)

    // (a) Pre-write cross-row validation inside the tx (so probes see same snapshot).
    if req.Body.Skills != nil {
        ids := make([][16]byte, 0, len(*req.Body.Skills))
        for _, s := range *req.Body.Skills {
            ids = append(ids, [16]byte(s.SkillId))
        }
        missing, mErr := qtx.SkillsMissingInOrg(ctx, generated.SkillsMissingInOrgParams{Column1: ids, OrgID: pgUUID(orgID)})
        if mErr != nil { /* 500 */ }
        if len(missing) > 0 {
            return api.UpdateAgent422JSONResponse{ErrorResponse: api.ErrorResponse{
                Error: api.ErrorCodeInvalidReference, Reason: fmt.Sprintf("unknown_skill_id:%s", uuid.UUID(missing[0])),
            }}, nil // tx auto-rollback via defer
        }
    }

    // (b) Atomic version-checked UPDATE inside the tx (Codex C3: sqlc.narg('name')/etc COALESCE).
    row, err := qtx.UpdateAgent(ctx, generated.UpdateAgentParams{
        ID:              pgUUID(agentID),
        OrgID:           pgUUID(orgID),
        ExpectedVersion: int32(req.Body.Version),
        Name:            nargString(req.Body.Name),       // sqlc.narg → COALESCE preserves on nil
        Email:           nargString((*string)(req.Body.Email)),
        Enabled:         nargBool(req.Body.Enabled),
    })
    if errors.Is(err, pgx.ErrNoRows) {
        // 0 rows — 404 vs 409 disambiguation. GetAgentByIdAnyVersion runs INSIDE the same tx.
        cur, perr := qtx.GetAgentByIdAnyVersion(ctx, generated.GetAgentByIdAnyVersionParams{ID: pgUUID(agentID), OrgID: pgUUID(orgID)})
        if errors.Is(perr, pgx.ErrNoRows) { return 404, nil }
        if perr != nil                    { return 500, nil }
        // 409 — defer Rollback runs automatically (no agent UPDATE landed).
        _ = h.deps.Cache.Del(ctx, cache.Key(orgID, "agents", agentID))  // D-56 cache DEL on 409
        return api.UpdateAgent409JSONResponse{Current: mapAgent(cur, nil), Error: "version_conflict", Reason: "version_mismatch"}, nil
    }
    if err != nil { /* 500 — defer Rollback */ }

    // (c) Skills replace INSIDE the same tx (uses qtx, NOT a nested BeginTx).
    if req.Body.Skills != nil {
        if _, err := qtx.DeleteAgentSkills(ctx, generated.DeleteAgentSkillsParams{AgentID: pgUUID(agentID), OrgID: pgUUID(orgID)}); err != nil {
            /* 500 — defer Rollback */
        }
        for _, sk := range *req.Body.Skills {
            if _, err := qtx.InsertAgentSkill(ctx, generated.InsertAgentSkillParams{
                AgentID: pgUUID(agentID), SkillID: pgUUID(uuid.UUID(sk.SkillId)),
                OrgID: pgUUID(orgID), Proficiency: int32(sk.Proficiency),
            }); err != nil { /* 500 — defer Rollback */ }
        }
    }

    // (d) Commit + cache DEL.
    if err := tx.Commit(ctx); err != nil { /* 500 */ }
    _ = h.deps.Cache.Del(ctx, cache.Key(orgID, "agents", agentID))  // D-55 after commit

    // (e) Load fresh skills for response.
    skills, _ := generated.New(h.deps.OrgDB).ListSkillsForAgent(ctx, generated.ListSkillsForAgentParams{AgentID: pgUUID(agentID), OrgID: pgUUID(orgID)})
    return api.UpdateAgent200JSONResponse(mapAgent(row, skills)), nil
}
```
