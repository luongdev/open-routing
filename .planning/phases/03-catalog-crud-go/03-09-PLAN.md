```go
// mapAdapter unmarshals []byte → map[string]any
func mapAdapter(row generated.Adapter) (api.Adapter, error) {
    var cfg map[string]any
    if len(row.Config) > 0 {
        if err := json.Unmarshal(row.Config, &cfg); err != nil {
            return api.Adapter{}, fmt.Errorf("config unmarshal: %w", err)
        }
    } else {
        cfg = map[string]any{}
    }
    return api.Adapter{
        Id:          api.UUIDv7(apiUUID(row.ID)),
        OrgId:       api.UUIDv7(apiUUID(row.OrgID)),
        Name:        row.Name,
        AdapterType: row.AdapterType,
        Config:      cfg,
        Enabled:     row.Enabled,
        Version:     int(row.Version),
        CreatedAt:   ptrTime(row.CreatedAt),
        UpdatedAt:   ptrTime(row.UpdatedAt),
    }, nil
}
```

GetAdapter cache loader uses `mapAdapter` and propagates the marshal error as a 500 from inside the loader (cache.GetOrSet returns the error).

For UpdateAdapter Update params: same json.Marshal of req.Body.Config (if supplied — `req.Body.Config` is `*map[string]any` nullable on UpdateAdapterRequest). If nil, do NOT update config; if non-nil, marshal and pass.

This means the UpdateAdapter sql query (Plan 03-03) must support either (a) one parameter for config that's always sent, or (b) `COALESCE(config, $4)` pattern with NULL passing to keep existing. Simpler: always send the new config bytes (if `req.Body.Config == nil`, marshal an empty map `{}` OR fall back to the current value via a GetAdapter call first). Recommendation: ALWAYS send the current config on PATCH — if user doesn't supply `config`, look up the existing row first and pass its config unchanged. This avoids a NULL-overwrite footgun:

```go
// In UpdateAdapter handler, BEFORE the UPDATE:
var cfgBytes []byte
if req.Body.Config != nil {
    cfgBytes, _ = json.Marshal(*req.Body.Config)
} else {
    // Preserve existing config — read it first.
    existing, _ := q.GetAdapter(ctx, generated.GetAdapterParams{ID: pgUUID(adapterID), OrgID: pgUUID(orgID)})
    cfgBytes = existing.Config
}
```

This adds an extra DB round-trip but keeps semantics intuitive ("PATCH without config = unchanged config"). Acceptable cost.

```go
package catalog

import (

// listSkillsForAgent loads the agent's skill assignments. Returns nil
// when the agent has no skills. Used by GetAgent's cache loader (CAT-01
// detail responses embed skills; CAT-10 list responses do NOT).
func (h *Handlers) listSkillsForAgent(ctx context.Context, agentID, orgID uuid.UUID) ([]api.AgentSkillAssignment, error) {
    q := generated.New(h.deps.OrgDB)
    rows, err := q.ListSkillsForAgent(ctx, generated.ListSkillsForAgentParams{
        AgentID: pgUUID(agentID), OrgID: pgUUID(orgID),
    })
    if err != nil {
        return nil, fmt.Errorf("list skills for agent: %w", err)
    }
    if len(rows) == 0 {
        return nil, nil
    }
    list := make([]api.AgentSkillAssignment, 0, len(rows))
    for _, r := range rows {
        name := r.SkillName
        list = append(list, api.AgentSkillAssignment{
            SkillId:     api.UUIDv7(apiUUID(r.SkillID)),
            Proficiency: int(r.Proficiency),
            Name:        &name,
        })
    }
    return list, nil
}

// validateSkillsExistInOrg runs the D-76 cross-row probe: every skill_id
// in `assignments` must exist + be enabled in the same org. Returns the
// first missing skill_id (for the 422 reason field) or nil on success.
//
// Note: the SQL function returns UUIDs as [16]byte from sqlc; we surface
// the first missing one. If callers need the full list, return a slice.
func (h *Handlers) validateSkillsExistInOrg(ctx context.Context, orgID uuid.UUID, assignments []api.AgentSkillAssignment) (*uuid.UUID, error) {
    if len(assignments) == 0 {
        return nil, nil
    }
    q := generated.New(h.deps.OrgDB)
    ids := make([][16]byte, 0, len(assignments))
    for _, s := range assignments {
        ids = append(ids, [16]byte(s.SkillId))
    }
    missing, err := q.SkillsMissingInOrg(ctx, generated.SkillsMissingInOrgParams{
        Column1: ids, OrgID: pgUUID(orgID),
    })
    if err != nil {
        return nil, fmt.Errorf("skills probe: %w", err)
    }
    if len(missing) > 0 {
        u := uuid.UUID(missing[0])
        return &u, nil
    }
    return nil, nil
}

// replaceAgentSkills runs DeleteAgentSkills + N InsertAgentSkill against
// an EXISTING enclosing tx (Codex C4 iter 3 — composes with the agent
// UPDATE/INSERT tx in agents.go so the agent row write and skills replace
// are atomic). Caller passes the per-tx Queries handle (`generated.New(tx)`),
// NOT the OrgDB.
//
// Renamed from the earlier `replaceAgentSkillsTx` (which owned its own
// BeginTx) — that signature only worked for standalone skills-only
// operations, but there's no standalone /agents/{id}/skills endpoint in
// the v0.1 spec. UpdateAgent + CreateAgent are the only callers, and they
// MUST share the agent row's tx (Codex C4 atomicity). The helper no longer
// suffixes "Tx" because composition with an enclosing tx is now the only
// pattern. Pitfall 5 (atomicity) holds — the enclosing handler's
// defer Rollback covers both the agent row write and the skills replace.
//
// Caller MUST have already validated skill_ids exist + proficiency is in
// range via validateSkillsExistInOrg + validateProficiencyRange before
// calling this helper. Codex C4 also requires the validation probes to run
// against the SAME tx (qtx) so they see the same snapshot — pass qtx in
// from the handler.
func (h *Handlers) replaceAgentSkills(ctx context.Context, qtx *generated.Queries, orgID, agentID uuid.UUID, assignments []api.AgentSkillAssignment) error {
    if _, err := qtx.DeleteAgentSkills(ctx, generated.DeleteAgentSkillsParams{
        AgentID: pgUUID(agentID), OrgID: pgUUID(orgID),
    }); err != nil {
        return fmt.Errorf("delete agent skills: %w", err)
    }
    for _, s := range assignments {
        if _, err := qtx.InsertAgentSkill(ctx, generated.InsertAgentSkillParams{
            AgentID:     pgUUID(agentID),
            SkillID:     pgUUID(uuid.UUID(s.SkillId)),
            OrgID:       pgUUID(orgID),
            Proficiency: int32(s.Proficiency),
        }); err != nil {
            return fmt.Errorf("insert agent skill %s: %w", uuid.UUID(s.SkillId), err)
        }
    }
    return nil  // commit is the caller's responsibility (the enclosing tx in agents.go).
}
```

Note: `*generated.Queries` is the type returned by `generated.New(dbtx)` from sqlc. Both `*OrgDB` and `*OrgTx` satisfy `generated.DBTX`, so `generated.New(tx)` works inside an `OrgDB.BeginTx`-opened tx; the SQLChecker (FOUND-04) is preserved across the tx scope.

```go
// validateProficiencyRange checks that every assignment's proficiency is
// in the inclusive range 1-10. Returns the FIRST out-of-range field path
// (e.g. "skills[2].proficiency") so the handler can surface it in the 422
// response body's `field` slot per the spec ErrorResponse envelope.
//
// Per ROADMAP Phase 3 success criterion 4 + Plan 03-01 spec amendment:
// Layer 1 schema does NOT carry min/max for proficiency (we removed it so
// oapi-codegen passes through and the handler returns 422 invalid_value
// instead of 400 invalid_body). Handlers MUST call this BEFORE the
// validateSkillsExistInOrg probe.
func validateProficiencyRange(assignments []api.AgentSkillAssignment) (badIdx int, ok bool) {
    for i, s := range assignments {
        if s.Proficiency < 1 || s.Proficiency > 10 {
            return i, false
        }
    }
    return 0, true
}
```
