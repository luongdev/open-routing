**`services/api/internal/db/queries/agents.sql` target (~120 LOC):**

```sql
-- name: InsertAgent :one
INSERT INTO agents (id, org_id, external_id, name, email, enabled)
VALUES ($1, $2, $3, $4, $5, $6)
RETURNING id, org_id, external_id, name, email, enabled, version, created_at, updated_at;

-- name: GetAgent :one
SELECT id, org_id, external_id, name, email, enabled, version, created_at, updated_at
FROM agents
WHERE id = $1 AND org_id = $2;

-- name: GetAgentByIdAnyVersion :one
-- Used for 404-vs-409 disambiguation when UpdateAgent returns 0 rows (D-66).
SELECT id, org_id, external_id, name, email, enabled, version, created_at, updated_at
FROM agents
WHERE id = $1 AND org_id = $2;

-- name: ListAgents :many
-- Default-list path: enabled = TRUE (D-65).
SELECT id, org_id, external_id, name, email, enabled, version, created_at, updated_at
FROM agents
WHERE org_id = $1
  AND enabled = TRUE
  AND (sqlc.narg('cursor_at')::timestamptz IS NULL OR (created_at, id) < (sqlc.narg('cursor_at'), sqlc.narg('cursor_id')))
  AND (sqlc.narg('name_filter')::text IS NULL OR lower(name) LIKE '%' || lower(sqlc.narg('name_filter')::text) || '%')
ORDER BY created_at DESC, id DESC
LIMIT $2;  -- N + 1 sentinel computed handler-side

-- name: ListAgentsIncludingDisabled :many
-- ?include_disabled=true path: omits enabled = TRUE filter (D-65).
SELECT id, org_id, external_id, name, email, enabled, version, created_at, updated_at
FROM agents
WHERE org_id = $1
  AND (sqlc.narg('cursor_at')::timestamptz IS NULL OR (created_at, id) < (sqlc.narg('cursor_at'), sqlc.narg('cursor_id')))
  AND (sqlc.narg('name_filter')::text IS NULL OR lower(name) LIKE '%' || lower(sqlc.narg('name_filter')::text) || '%')
ORDER BY created_at DESC, id DESC
LIMIT $2;

-- name: UpdateAgent :one
-- Atomic version-checked UPDATE (D-66) with sparse-PATCH semantics (Codex C3 iter 3).
-- COALESCE(sqlc.narg('field'), field) preserves the current column value when the
-- handler passes a NULL — this is REQUIRED for PATCH semantics where the client
-- may omit fields (e.g. "PATCH /agents/{id} with only {version, name}" must NOT
-- zero-out email/enabled). 0 rows → handler issues a follow-up GetAgentByIdAnyVersion
-- to disambiguate 404 (no row) vs 409 (row exists, version mismatched).
UPDATE agents
SET name       = COALESCE(sqlc.narg('name'),    name),
    email      = COALESCE(sqlc.narg('email'),   email),
    enabled    = COALESCE(sqlc.narg('enabled'), enabled),
    version    = version + 1,
    updated_at = NOW()
WHERE id = sqlc.arg('id')
  AND org_id = sqlc.arg('org_id')
  AND version = sqlc.arg('expected_version')
RETURNING id, org_id, external_id, name, email, enabled, version, created_at, updated_at;

-- name: SoftDeleteAgent :execrows
-- Idempotent soft delete (D-65). WHERE enabled = TRUE means re-deleting an
-- already-soft-deleted row returns 0 rows → 404 to client.
UPDATE agents SET enabled = FALSE, updated_at = NOW()
WHERE id = $1 AND org_id = $2 AND enabled = TRUE;
```

**`services/api/internal/db/orgdb.go` extension** — Add `BeginTx` (OQ-5):

```go
// OrgTx wraps pgx.Tx with the SQL validator so transactions preserve the
// FOUND-04 isolation guarantee. Satisfies generated.DBTX.
type OrgTx struct {
    tx      pgx.Tx
    checker *SQLChecker
    mode    ValidationMode
}

// Compile-time guarantee: OrgTx satisfies the sqlc-generated DBTX interface.
var _ generated.DBTX = (*OrgTx)(nil)

func (t *OrgTx) Exec(ctx context.Context, sql string, args ...interface{}) (pgconn.CommandTag, error) {
    if err := t.preflight(ctx, sql); err != nil { return pgconn.CommandTag{}, err }
    return t.tx.Exec(ctx, sql, args...)
}
func (t *OrgTx) Query(ctx context.Context, sql string, args ...interface{}) (pgx.Rows, error) {
    if err := t.preflight(ctx, sql); err != nil { return nil, err }
    return t.tx.Query(ctx, sql, args...)
}
func (t *OrgTx) QueryRow(ctx context.Context, sql string, args ...interface{}) pgx.Row {
    if err := t.preflight(ctx, sql); err != nil {
        // pgx.Row interface returns errors via Scan; we wrap our checker error.
        return errRow{err: err}
    }
    return t.tx.QueryRow(ctx, sql, args...)
}
func (t *OrgTx) Commit(ctx context.Context) error   { return t.tx.Commit(ctx) }
func (t *OrgTx) Rollback(ctx context.Context) error { return t.tx.Rollback(ctx) }

// preflight reuses OrgDB.preflight's logic.
func (t *OrgTx) preflight(ctx context.Context, sql string) error {
    return preflightSQL(ctx, sql, t.checker, t.mode)
}

// errRow returns the saved error on Scan; used when preflight rejects QueryRow
// SQL (pgx.Row has no Err method).
type errRow struct{ err error }
func (e errRow) Scan(...any) error { return e.err }

// BeginTx starts a transaction; the returned OrgTx satisfies generated.DBTX
// so handlers can pass it to generated.New(tx) (OQ-5, H5 agent_skills replace).
func (o *OrgDB) BeginTx(ctx context.Context) (*OrgTx, error) {
    tx, err := o.pool.BeginTx(ctx, pgx.TxOptions{})
    if err != nil { return nil, err }
    return &OrgTx{tx: tx, checker: o.checker, mode: o.mode}, nil
}
```
