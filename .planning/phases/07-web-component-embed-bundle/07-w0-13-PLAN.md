## Step 1: or-card.ts

```ts
import { LitElement, html, nothing } from 'lit';
import { customElement, property } from 'lit/decorators.js';

@customElement('or-card')
export class OrCard extends LitElement {
  @property({ type: Boolean }) hoverable = false;
  @property({ type: String }) variant: 'default' | 'primary' | 'secondary' = 'default';
  @property({ type: String }) padding: 'sm' | 'md' | 'lg' = 'md';

  override createRenderRoot() { return this; }

  private get classes(): string {
    return ['uk-card', `uk-card-${this.variant}`,
      this.padding === 'sm' ? 'uk-card-small' : this.padding === 'lg' ? 'uk-card-large' : '',
      this.hoverable && 'uk-card-hover'].filter(Boolean).join(' ');
  }

  override render() {
    const hasHeader = !!this.querySelector('[slot="header"]');
    const hasFooter = !!this.querySelector('[slot="footer"]');
    return html`
      <div class=${this.classes}>
        ${hasHeader ? html`<div class="uk-card-header"><slot name="header"></slot></div>` : nothing}
        <div class="uk-card-body"><slot></slot></div>
        ${hasFooter ? html`<div class="uk-card-footer"><slot name="footer"></slot></div>` : nothing}
      </div>
    `;
  }
}
```
