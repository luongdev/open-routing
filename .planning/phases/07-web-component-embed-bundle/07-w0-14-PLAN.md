## Step 1: or-badge.ts

```ts
import { LitElement, html, nothing } from 'lit';
import { customElement, property } from 'lit/decorators.js';

export type BadgeVariant = 'default' | 'success' | 'warning' | 'destructive' | 'info' | 'primary';

@customElement('or-badge')
export class OrBadge extends LitElement {
  @property({ type: String }) variant: BadgeVariant = 'default';
  @property({ type: Boolean }) pill = false;
  @property({ type: Boolean }) dot = false;

  override createRenderRoot() { return this; }

  private get classes(): string {
    return ['uk-label', `uk-label-${this.variant}`,
      this.pill && 'uk-label-pill', this.dot && 'uk-label-dot'].filter(Boolean).join(' ');
  }

  override render() {
    return html`
      <span class=${this.classes} role="status">
        ${this.dot ? nothing : html`<slot></slot>`}
      </span>
    `;
  }
}
```

(If `uk-label-pill` / `uk-label-dot` classes don't exist in Frankenstyle,
write custom styles in `or-badge.css` mapping pill → `border-radius: 9999px`
and dot → `width: 8px; height: 8px; border-radius: 50%`. Light DOM
allows pure CSS in a sibling file imported via `franken.css`.)
