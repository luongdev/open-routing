## Step 1: or-sidebar.ts

```ts
import { LitElement, html, nothing } from 'lit';
import { customElement, property } from 'lit/decorators.js';
import './or-icon.js';

export interface SidebarItem {
  id: string;
  label: string;
  icon: string;
  href: string;
  badge?: string | number;
}
export interface SidebarSection {
  id: string;
  label: string;
  items: SidebarItem[];
}

@customElement('or-sidebar')
export class OrSidebar extends LitElement {
  @property({ type: Array }) sections: SidebarSection[] = [];
  @property({ type: String, attribute: 'active-item-id' }) activeItemId = '';
  @property({ type: Boolean, reflect: true }) collapsed = false;

  override createRenderRoot() { return this; }

  private onSelect(item: SidebarItem, e: Event): void {
    e.preventDefault();
    this.dispatchEvent(new CustomEvent('or-nav-select', {
      detail: { id: item.id, href: item.href }, bubbles: true, composed: true,
    }));
  }

  override render() {
    return html`
      <aside class="or-sidebar ${this.collapsed ? 'or-sidebar-collapsed' : ''}" aria-label="Navigation">
        <header class="or-sidebar-header">
          <slot name="logo"></slot>
        </header>
        <nav>
          ${this.sections.map(section => html`
            <div class="or-sidebar-section">
              ${this.collapsed ? nothing : html`<div class="or-sidebar-section-label">${section.label}</div>`}
              <ul role="list">
                ${section.items.map(item => html`
                  <li>
                    <a
                      href=${item.href}
                      class=${item.id === this.activeItemId ? 'or-nav-item or-nav-active' : 'or-nav-item'}
                      aria-current=${item.id === this.activeItemId ? 'page' : 'false'}
                      @click=${(e: Event) => this.onSelect(item, e)}
                    >
                      <or-icon name=${item.icon} size="16"></or-icon>
                      ${this.collapsed ? nothing : html`
                        <span class="or-nav-label">${item.label}</span>
                        ${item.badge !== undefined ? html`<span class="or-nav-badge">${item.badge}</span>` : nothing}
                      `}
                    </a>
                  </li>
                `)}
              </ul>
            </div>
          `)}
        </nav>
        <footer class="or-sidebar-footer">
          <slot name="user-menu"></slot>
        </footer>
      </aside>
    `;
  }
}
```
