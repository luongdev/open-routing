## Step 1: or-dialog.ts

```ts
import { LitElement, html } from 'lit';
import { customElement, property } from 'lit/decorators.js';
// <uk-modal> is auto-registered by hwc-bootstrap.ts from W0.0-01.

@customElement('or-dialog')
export class OrDialog extends LitElement {
  @property({ type: Boolean, reflect: true }) open = false;
  @property({ type: String }) size: 'sm' | 'md' | 'lg' | 'xl' = 'md';
  @property({ type: Boolean, attribute: 'prevent-close' }) preventClose = false;

  override createRenderRoot() { return this; }

  private onOpen(): void {
    this.dispatchEvent(new CustomEvent('or-dialog-open', { bubbles: true, composed: true }));
  }
  private onClose(): void {
    this.open = false;
    this.dispatchEvent(new CustomEvent('or-dialog-close', { bubbles: true, composed: true }));
  }

  override render() {
    return html`
      <uk-modal
        ?open=${this.open}
        size=${this.size}
        ?keyboard=${!this.preventClose}
        ?bg-close=${!this.preventClose}
        @uk-show=${this.onOpen}
        @uk-hide=${this.onClose}
      >
        <div class="uk-modal-dialog">
          <header class="uk-modal-header"><slot name="header"></slot></header>
          <div class="uk-modal-body"><slot></slot></div>
          <footer class="uk-modal-footer"><slot name="footer"></slot></footer>
        </div>
      </uk-modal>
    `;
  }
}
```

(Verify exact `<uk-modal>` attribute API by reading Frankenstyle docs at
https://franken.style/docs/latest/kit/modal — adjust attribute names if
Frankenstyle uses different names than `keyboard` / `bg-close`.)
