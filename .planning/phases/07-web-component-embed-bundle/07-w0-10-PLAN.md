<objective>
Build `<or-button>` — first redesigned primitive. Wraps Frankenstyle's
`.uk-btn` classes in a Lit component for type-safe props. Light DOM
(no Shadow DOM) so Frankenstyle classes resolve from global styles.

**Why light DOM**: Frankenstyle styles `.uk-btn` globally. Shadow DOM
isolation would block those styles. Lit lets us opt out per-component
by overriding `createRenderRoot()`. This is the documented pattern for
class-based design systems with Lit (see Lit docs: "Working with
Shadow DOM" → "Skip shadow DOM").

**Out of scope:**
- Touching existing sl-button usages (W0.0-30)
- Other variants like icon-button (add later if needed)
- Theming (W0.0-02 owns palette; this plan reads vars, doesn't define them)
</objective>

## Step 1: Write `or-button.ts`

```ts
import { LitElement, html, css, nothing } from 'lit';
import { customElement, property } from 'lit/decorators.js';

export type ButtonVariant = 'default' | 'primary' | 'secondary' | 'ghost' | 'destructive';
export type ButtonSize = 'sm' | 'md' | 'lg';

@customElement('or-button')
export class OrButton extends LitElement {
  @property({ type: String }) variant: ButtonVariant = 'default';
  @property({ type: String }) size: ButtonSize = 'md';
  @property({ type: Boolean }) disabled = false;
  @property({ type: Boolean }) loading = false;
  @property({ type: String }) type: 'button' | 'submit' | 'reset' = 'button';

  // Light DOM — Frankenstyle uk-btn classes resolve from global stylesheet
  override createRenderRoot() {
    return this;
  }

  private get classNames(): string {
    const base = 'uk-btn';
    const variantClass = `uk-btn-${this.variant}`;
    const sizeClass = this.size === 'md' ? '' : `uk-btn-${this.size}`;
    return [base, variantClass, sizeClass].filter(Boolean).join(' ');
  }

  private onClick(e: Event): void {
    if (this.loading) {
      e.preventDefault();
      e.stopImmediatePropagation();
    }
  }

  override render() {
    return html`
      <button
        type=${this.type}
        class=${this.classNames}
        ?disabled=${this.disabled || this.loading}
        @click=${this.onClick}
        aria-busy=${this.loading ? 'true' : 'false'}
      >
        ${this.loading ? html`<span class="uk-spinner uk-spinner-sm" aria-hidden="true"></span>` : nothing}
        <slot></slot>
      </button>
    `;
  }
}

declare global {
  interface HTMLElementTagNameMap {
    'or-button': OrButton;
  }
}
```

## Step 2: Write `or-button.test.ts`

Cover: variant class mapping, size mapping, disabled, loading suppresses click, slot content renders, type attribute. ~6 tests.

```ts
import { describe, it, expect, beforeEach } from 'vitest';
import './or-button.js';

describe('OrButton', () => {
  let el: HTMLElement;
  beforeEach(() => {
    el = document.createElement('or-button');
    document.body.appendChild(el);
  });

  it('default variant + md size → uk-btn uk-btn-default', async () => {
    await (el as any).updateComplete;
    expect(el.querySelector('button')?.className).toBe('uk-btn uk-btn-default');
  });

  it('variant=primary, size=lg → uk-btn uk-btn-primary uk-btn-lg', async () => {
    (el as any).variant = 'primary';
    (el as any).size = 'lg';
    await (el as any).updateComplete;
    expect(el.querySelector('button')?.className).toBe('uk-btn uk-btn-primary uk-btn-lg');
  });

  it('disabled propagates to button', async () => {
    (el as any).disabled = true;
    await (el as any).updateComplete;
    expect((el.querySelector('button') as HTMLButtonElement).disabled).toBe(true);
  });

  it('loading shows spinner + disables button + suppresses click', async () => {
    (el as any).loading = true;
    await (el as any).updateComplete;
    expect(el.querySelector('.uk-spinner')).toBeTruthy();
    expect((el.querySelector('button') as HTMLButtonElement).disabled).toBe(true);
  });

  it('slot content renders', async () => {
    el.innerHTML = 'Hello';
    await (el as any).updateComplete;
    expect(el.textContent).toContain('Hello');
  });

  it('type=submit on button', async () => {
    (el as any).type = 'submit';
    await (el as any).updateComplete;
    expect((el.querySelector('button') as HTMLButtonElement).type).toBe('submit');
  });
});
```
