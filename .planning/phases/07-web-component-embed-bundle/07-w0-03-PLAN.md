<objective>
Build the UI Playground — a single route in admin app that renders all
13 component categories in a labelled grid. Each downstream W0.0
component plan (10..22) fills one slot. Playwright screenshot diff
against the Ember reference PNGs is the self-verification step every
downstream plan uses.

**Why now (not after components)**: The playground is the verification
contract for every component plan. Without it, AI implementing W0.0-10
(or-button) has nowhere to render the new component in isolation —
they'd have to deploy to a real page, which couples the component
build with migration. With the playground, each plan is:
1. Build component
2. Drop into playground slot
3. Screenshot vs Ember reference
4. Done

**Out of scope:**
- Building any actual component (downstream plans)
- Migrating any existing component (07-w0-30)
- Storybook-style props panel (not needed; static showcase is enough)
- Auto-screenshot harness in CI (deferred; AI runs Playwright locally per plan)
</objective>

## Step 3: Build `playground-route.ts`

```ts
import { LitElement, html, css } from 'lit';
import { customElement } from 'lit/decorators.js';
// <uk-theme-switcher> is auto-registered by hwc-bootstrap.ts from W0.0-01.
// No import needed here.

@customElement('or-playground-route')
export class OrPlaygroundRoute extends LitElement {
  static override styles = css`
    :host {
      display: block;
      padding: 32px;
      background: var(--background);
      color: var(--foreground);
      min-height: 100vh;
    }
    .header {
      display: flex;
      justify-content: space-between;
      align-items: center;
      margin-bottom: 32px;
    }
    .grid {
      display: grid;
      grid-template-columns: repeat(auto-fill, minmax(320px, 1fr));
      gap: 24px;
    }
    .slot {
      padding: 24px;
      border: 1px solid var(--border);
      border-radius: 12px;
      background: var(--card);
    }
    .slot h3 {
      margin: 0 0 16px;
      font-size: 14px;
      font-weight: 600;
      color: var(--muted-foreground);
      text-transform: uppercase;
      letter-spacing: 0.05em;
    }
    .slot .empty {
      color: var(--muted-foreground);
      font-style: italic;
      padding: 24px;
      text-align: center;
      border: 2px dashed var(--border);
      border-radius: 8px;
    }
  `;

  override render() {
    return html`
      <div class="header">
        <h1>UI Playground</h1>
        <uk-theme-switcher></uk-theme-switcher>
      </div>
      <div class="grid">
        ${SLOTS.map(
          (s) => html`
            <section class="slot" data-component=${s.id}>
              <h3>${s.label}</h3>
              <div class="empty">⏳ ${s.plan} pending</div>
            </section>
          `
        )}
      </div>
    `;
  }
}

const SLOTS = [
  { id: 'button', label: 'Button', plan: '07-w0-10' },
  { id: 'input', label: 'Input', plan: '07-w0-11' },
  { id: 'select', label: 'Select', plan: '07-w0-12' },
  { id: 'card', label: 'Card', plan: '07-w0-13' },
  { id: 'badge', label: 'Badge', plan: '07-w0-14' },
  { id: 'dialog', label: 'Dialog', plan: '07-w0-15' },
```

## Step 5: Write `playground-route.test.ts`

```ts
import { describe, it, expect, beforeEach } from 'vitest';
import './playground-route.js';

describe('OrPlaygroundRoute', () => {
  let el: HTMLElement;

  beforeEach(() => {
    el = document.createElement('or-playground-route');
    document.body.appendChild(el);
  });

  it('renders 13 slots', async () => {
    await (el as any).updateComplete;
    const slots = el.shadowRoot!.querySelectorAll('section.slot');
    expect(slots.length).toBe(13);
  });

  it('renders theme switcher in header', async () => {
    await (el as any).updateComplete;
    expect(el.shadowRoot!.querySelector('uk-theme-switcher')).toBeTruthy();
  });

  it.each([
    ['button', '07-w0-10'],
    ['table', '07-w0-19'],
    ['sidebar', '07-w0-22'],
  ])('slot %s references plan %s', async (id, plan) => {
    await (el as any).updateComplete;
    const slot = el.shadowRoot!.querySelector(`[data-component="${id}"]`);
    expect(slot?.textContent).toContain(plan);
  });
});
```
