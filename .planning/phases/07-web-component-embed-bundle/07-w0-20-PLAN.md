<objective>
Imperative toast API via `notify()` — wraps Frankenstyle's
`UkNotification.notification(message, options)`. Plus declarative
`<or-toast>` element for static toasts in markup if needed.

**Why imperative**: Toasts are fundamentally side-effects (e.g.,
"Save successful"). API errors from anywhere in the app should call
`notifyError(err)` without rendering a component.
</objective>

## Step 1: notify.ts

```ts
// <uk-notification> is auto-registered by hwc-bootstrap.ts from W0.0-01.
// Frankenstyle exposes `window.UIkit.notification(...)` imperative API
// after hwc-bootstrap loads. Verify exact namespace by inspecting
// `window.UIkit` in browser console after install.

export type NotifyVariant = 'info' | 'success' | 'warning' | 'destructive';
export type NotifyPosition = 'top-left' | 'top-center' | 'top-right' | 'bottom-left' | 'bottom-center' | 'bottom-right';

export interface NotifyOptions {
  message: string;
  variant?: NotifyVariant;
  duration?: number;
  position?: NotifyPosition;
}

export function notify(opts: NotifyOptions): void {
  const Uk = (window as any).UIkit;
  if (!Uk?.notification) {
    console.error('Frankenstyle notification API not loaded');
    return;
  }
  Uk.notification({
    message: opts.message,
    status: opts.variant === 'destructive' ? 'danger' : opts.variant ?? 'info',
    pos: opts.position ?? 'top-right',
    timeout: opts.duration ?? 5000,
  });
}

export function notifyError(err: unknown): void {
  const message = err instanceof Error ? err.message
    : typeof err === 'string' ? err
    : 'Something went wrong';
  notify({ message, variant: 'destructive' });
}
```

(If Frankenstyle exposes `<uk-notification>` element instead of
imperative API — check Frankenstyle docs at
https://franken.style/docs/latest/kit/notification — swap above for
declarative `document.body.appendChild(el)` pattern.)
