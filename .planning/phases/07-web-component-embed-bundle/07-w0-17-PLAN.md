<objective>
`<or-icon>` — type-safe wrapper around Frankenstyle's built-in `<uk-icon>`
element (lucide icon set ships as a dependency of frankenstyle —
already loaded via hwc-bootstrap.ts in W0.0-01).

**Why wrap uk-icon (not raw)**: TypeScript union enforces only valid
icon names. Catches typos at compile time. Same pattern as shadcn's
typed Lucide imports.

**Why not separate sprite (original plan revision)**: Original plan
proposed installing `lucide-static` and building a custom sprite.
Unnecessary — Frankenstyle bundles lucide. Use `<uk-icon icon="home">`
directly.
</objective>

<execution_context>
Frankenstyle ships `lucide@^0.562.0` as a runtime dep (per its
package.json). `<uk-icon>` Custom Element reads `icon` attribute and
renders the matching lucide SVG inline. No sprite, no extra requests.

Frankenstyle `<uk-icon>` attribute API (verify against
https://franken.style/docs/latest/kit/icon if differs):
- `icon` — lucide icon name (kebab-case)
- `height` / `width` — pixel dimensions
- `stroke-width` — optional, default 2
</execution_context>

## Step 2: `or-icon.ts`

```ts
import { LitElement, html, css } from 'lit';
import { customElement, property } from 'lit/decorators.js';
import { ICON_NAMES, type IconName } from './icon-names.js';

@customElement('or-icon')
export class OrIcon extends LitElement {
  @property({ type: String }) name: IconName | '' = '';
  @property({ type: Number }) size = 16;
  @property({ type: String }) label = '';

  static override styles = css`
    :host { display: inline-flex; vertical-align: middle; line-height: 0; }
  `;

  override render() {
    if (this.name && !ICON_NAMES.includes(this.name as IconName)) {
      console.warn(`<or-icon> unknown name: ${this.name}`);
      return null;
    }
    return html`
      <uk-icon
        icon=${this.name}
        height=${this.size}
        width=${this.size}
        aria-hidden=${this.label ? 'false' : 'true'}
        aria-label=${this.label || undefined}
        role=${this.label ? 'img' : undefined}
      ></uk-icon>
    `;
  }
}

declare global {
  interface HTMLElementTagNameMap {
    'or-icon': OrIcon;
  }
}
```

## Step 3: Tests

```ts
import { describe, it, expect, beforeEach, vi } from 'vitest';
import './or-icon.js';

describe('OrIcon', () => {
  let el: HTMLElement;
  beforeEach(() => {
    el = document.createElement('or-icon');
    document.body.appendChild(el);
  });

  it('renders <uk-icon> with icon attribute', async () => {
    (el as any).name = 'home';
    await (el as any).updateComplete;
    const inner = el.shadowRoot!.querySelector('uk-icon');
    expect(inner?.getAttribute('icon')).toBe('home');
  });

  it('size sets height/width', async () => {
    (el as any).name = 'home';
    (el as any).size = 24;
    await (el as any).updateComplete;
    const inner = el.shadowRoot!.querySelector('uk-icon');
    expect(inner?.getAttribute('height')).toBe('24');
    expect(inner?.getAttribute('width')).toBe('24');
  });

  it('label sets aria-label + role=img', async () => {
    (el as any).name = 'home';
    (el as any).label = 'Dashboard';
    await (el as any).updateComplete;
    const inner = el.shadowRoot!.querySelector('uk-icon');
    expect(inner?.getAttribute('aria-label')).toBe('Dashboard');
    expect(inner?.getAttribute('role')).toBe('img');
  });

  it('absent label → aria-hidden=true', async () => {
    (el as any).name = 'home';
    await (el as any).updateComplete;
    const inner = el.shadowRoot!.querySelector('uk-icon');
    expect(inner?.getAttribute('aria-hidden')).toBe('true');
  });

  it('unknown name → console.warn + null render', async () => {
    const warn = vi.spyOn(console, 'warn').mockImplementation(() => {});
    (el as any).name = 'totally-fake';
    await (el as any).updateComplete;
    expect(warn).toHaveBeenCalled();
    warn.mockRestore();
  });
});
```
